# Phase 3 — IPC admin et flux d'events : plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `irisd` sert un pipe admin JSON-RPC (`\\.\pipe\iris-admin`), un flux SSE (`127.0.0.1:8899/events`) et démarre coffre fermé, le MITM s'activant au déverrouillage par console ou par `vault.unlock`.

**Architecture:** Une `VaultGate` détient l'état du coffre et publie secrets et CA d'un coup ; le `ProxyServer` est construit verrouillé et activé par la porte, sans redémarrer l'écoute. Le pipe (`AdminServer`, P/Invoke `CreateNamedPipe`) délègue à un `AdminDispatcher` qui route vers des gestionnaires par domaine ; `EventsServer` réutilise `HttpParser` sur un `TcpListener`. `irisd` câble le tout dans `DaemonServices` et garde l'invite console dans `ConsoleUnlock`.

**Tech Stack:** C# 5, .NET Framework 4.8, `csc.exe` natif, `System.IO.Pipes`, P/Invoke kernel32, `KeePassLib` (dans `KeePass.exe` 2.61.1), harnais de tests maison (`tests\Check.cs`).

**Spec:** `docs/superpowers/specs/2026-10-18-phase-3-ipc-design.md` (s'appuie sur `docs/superpowers/specs/2026-10-17-socle-design.md` et `../Iris/SPECS.md` §13, §14).

```csharp
using System;
using System.IO;
using System.Threading;
using System.Threading.Tasks;

namespace IrisKit
{
    /// <summary>Trame annonçant plus que le plafond autorisé : la connexion doit être fermée sans réponse.</summary>
    public sealed class FrameTooLargeException : IOException
    {
        public long Declared { get; private set; }
        public int Maximum { get; private set; }

        public FrameTooLargeException(long declared, int maximum)
            : base("trame de " + declared + " octets refusée (maximum " + maximum + ")")
        {
            Declared = declared;
            Maximum = maximum;
        }
    }

    /// <summary>Trames du pipe admin (SPECS §13.1) : 4 octets big-endian (longueur du JSON) puis le JSON UTF-8.</summary>
    public static class FrameCodec
    {
        public const int MaxRequestBytes = 1 << 20;
        public const int MaxResponseBytes = 16 << 20;

        /// <summary>Une trame complète ; null si le flux se ferme avant tout octet ; EndOfStreamException si elle est tronquée.</summary>
        public static async Task<byte[]> ReadAsync(Stream stream, int maxBytes, CancellationToken cancel)
        {
            byte[] header = new byte[4];
            int got = await FillAsync(stream, header, 0, 4, cancel).ConfigureAwait(false);
            if (got == 0) return null;
            if (got < 4) throw new EndOfStreamException("en-tête de trame tronqué");
            long length = ((long)header[0] << 24) | ((long)header[1] << 16) | ((long)header[2] << 8) | header[3];
            if (length > maxBytes) throw new FrameTooLargeException(length, maxBytes);
            byte[] payload = new byte[length];
            int read = await FillAsync(stream, payload, 0, (int)length, cancel).ConfigureAwait(false);
            if (read < length)
            {
                Array.Clear(payload, 0, read);
                throw new EndOfStreamException("trame tronquée : " + read + "/" + length + " octets");
            }
            return payload;
        }

        /// <summary>Écrit une trame et vide le flux ; la copie locale (qui peut porter une valeur de secret) est effacée.</summary>
        public static async Task WriteAsync(Stream stream, byte[] payload, CancellationToken cancel)
        {
            byte[] frame = new byte[4 + payload.Length];
            frame[0] = (byte)(payload.Length >> 24);
            frame[1] = (byte)(payload.Length >> 16);
            frame[2] = (byte)(payload.Length >> 8);
            frame[3] = (byte)payload.Length;
            Buffer.BlockCopy(payload, 0, frame, 4, payload.Length);
            try
            {
                await stream.WriteAsync(frame, 0, frame.Length, cancel).ConfigureAwait(false);
                await stream.FlushAsync(cancel).ConfigureAwait(false);
            }
            finally { Array.Clear(frame, 0, frame.Length); }
        }

        private static async Task<int> FillAsync(Stream stream, byte[] buffer, int offset, int count, CancellationToken cancel)
        {
            int total = 0;
            while (total < count)
            {
                int n = await stream.ReadAsync(buffer, offset + total, count - total, cancel).ConfigureAwait(false);
                if (n == 0) break;
                total += n;
            }
            return total;
        }
    }
}
```

```csharp
using System;
using System.Collections;
using System.Collections.Generic;
using System.Collections.Specialized;
using System.Text;

namespace IrisKit
{
    /// <summary>Erreur JSON-RPC (SPECS §13.2) : code standard, IRIS ou irispc, message, données optionnelles. Jamais de valeur de secret.</summary>
    public sealed class RpcException : Exception
    {
        public int Code { get; private set; }
        public object ErrorData { get; private set; }

        public RpcException(int code, string message) : this(code, message, null) { }

        public RpcException(int code, string message, object data) : base(message)
        {
            Code = code;
            ErrorData = data;
        }
    }

    /// <summary>Codes d'erreur : JSON-RPC 2.0, IRIS (-32001…-32012) et irispc (-32020…-32024, spec phase 3 §6).</summary>
    public static class RpcErrors
    {
        public const int ParseError = -32700;
        public const int InvalidRequest = -32600;
        public const int MethodNotFound = -32601;
        public const int InvalidParams = -32602;
        public const int InternalError = -32603;
        public const int UnknownSecret = -32001;
        public const int InvalidName = -32002;
        public const int InvalidAllowedHosts = -32003;
        public const int Duplicate = -32004;
        public const int DaemonPaused = -32005;
        public const int NotFound = -32006;
        public const int RuleProtected = -32010;
        public const int RuleNotFound = -32011;
        public const int ConfigReloadFailed = -32012;
        public const int VaultLocked = -32020;
        public const int BadPassword = -32021;
        public const int VaultBusy = -32022;
        public const int RateLimited = -32023;
        public const int VaultExists = -32024;
    }

    /// <summary>Requête décodée. `Id` : long, string ou null (JSON-RPC 2.0) ; `Params` : null si absent.</summary>
    public sealed class RpcRequest
    {
        public string Method { get; private set; }
        public JsonObject Params { get; private set; }
        public object Id { get; private set; }

        public RpcRequest(string method, JsonObject parameters, object id)
        {
            Method = method;
            Params = parameters;
            Id = id;
        }
    }

    /// <summary>Enveloppes JSON-RPC 2.0 du pipe admin, côté serveur et côté client.</summary>
    public static class JsonRpc
    {
        public const string Version = "2.0";
        private static readonly UTF8Encoding StrictUtf8 = new UTF8Encoding(false, true);

        /// <summary>
        /// Décode une requête. JSON ou UTF-8 illisible → -32700 ; enveloppe invalide → -32600 ; params non objet → -32602.
        /// `id` reçoit l'identifiant dès qu'il est lisible, pour que la réponse d'erreur le porte.
        /// </summary>
        public static RpcRequest ParseRequest(byte[] payload, out object id)
        {
            id = null;
            string text;
            try { text = StrictUtf8.GetString(payload); }
            catch (DecoderFallbackException) { throw new RpcException(RpcErrors.ParseError, "Parse error: UTF-8 invalide"); }
            object parsed;
            try { parsed = Json.Parse(text); }
            catch (JsonException ex) { throw new RpcException(RpcErrors.ParseError, "Parse error: " + ex.Message); }
            IDictionary<string, object> dict = parsed as IDictionary<string, object>;
            if (dict == null) throw new RpcException(RpcErrors.InvalidRequest, "Invalid Request: objet JSON attendu");
            object rawId;
            if (!dict.TryGetValue("id", out rawId)) throw new RpcException(RpcErrors.InvalidRequest, "Invalid Request: id manquant");
            if (rawId != null && !(rawId is long) && !(rawId is string))
                throw new RpcException(RpcErrors.InvalidRequest, "Invalid Request: id doit être entier, chaîne ou null");
            id = rawId;
            object version;
            dict.TryGetValue("jsonrpc", out version);
            if (!(version is string) || (string)version != Version)
                throw new RpcException(RpcErrors.InvalidRequest, "Invalid Request: jsonrpc doit valoir \"2.0\"");
            object method;
            dict.TryGetValue("method", out method);
            string name = method as string;
            if (string.IsNullOrEmpty(name)) throw new RpcException(RpcErrors.InvalidRequest, "Invalid Request: method manquante");
            object rawParams;
            dict.TryGetValue("params", out rawParams);
            JsonObject parameters = null;
            if (rawParams != null)
            {
                IDictionary<string, object> p = rawParams as IDictionary<string, object>;
                if (p == null) throw new RpcException(RpcErrors.InvalidParams, "Invalid params: objet attendu");
                parameters = new JsonObject(p);
            }
            return new RpcRequest(name, parameters, rawId);
        }

        public static byte[] Success(object id, object result)
        {
            OrderedDictionary o = new OrderedDictionary();
            o["jsonrpc"] = Version;
            o["id"] = id;
            o["result"] = result;
            return Encoding.UTF8.GetBytes(Json.EmitCompact(o));
        }

        public static byte[] Failure(object id, RpcException error)
        {
            OrderedDictionary e = new OrderedDictionary();
            e["code"] = (long)error.Code;
            e["message"] = error.Message;
            if (error.ErrorData != null) e["data"] = error.ErrorData;
            OrderedDictionary o = new OrderedDictionary();
            o["jsonrpc"] = Version;
            o["id"] = id;
            o["error"] = e;
            return Encoding.UTF8.GetBytes(Json.EmitCompact(o));
        }

        /// <summary>Requête côté client ; `parameters` null = pas de champ params.</summary>
        public static byte[] Request(long id, string method, IDictionary parameters)
        {
            OrderedDictionary o = new OrderedDictionary();
            o["jsonrpc"] = Version;
            o["method"] = method;
            if (parameters != null) o["params"] = parameters;
            o["id"] = id;
            return Encoding.UTF8.GetBytes(Json.EmitCompact(o));
        }

        /// <summary>Côté client : rend `result` (dictionnaire, liste ou scalaire) ou lève RpcException avec le code et le message du serveur.</summary>
        public static object ParseResponse(byte[] payload, long expectedId)
        {
            JsonObject o;
            try { o = Json.ParseObject(StrictUtf8.GetString(payload)); }
            catch (JsonException ex) { throw new RpcException(RpcErrors.ParseError, "réponse illisible : " + ex.Message); }
            catch (DecoderFallbackException) { throw new RpcException(RpcErrors.ParseError, "réponse illisible : UTF-8 invalide"); }
            JsonObject error = o.OptionalObject("error");
            if (error != null) throw new RpcException((int)error.Int("code"), error.String("message"), error.Raw("data"));
            object id = o.Raw("id");
            if (!(id is long) || (long)id != expectedId) throw new RpcException(RpcErrors.InvalidRequest, "réponse pour un autre id");
            if (!o.Has("result")) throw new RpcException(RpcErrors.InvalidRequest, "réponse sans result ni error");
            return o.Raw("result");
        }
    }
}
```

```csharp
        public static readonly string[] HotKeys = { "security.on_exfil_attempt", "security.max_substitutions_per_minute", "backups.max_count" };

        /// <summary>Applique le lot à `current` et rend la nouvelle config, non validée (Save valide). Clé inconnue → ConfigKeyException ; valeur non convertible → ConfigException.</summary>
        public static Config Apply(Config current, IList<KeyValuePair<string, string>> updates, out ConfigSetResult result)
        {
            BrokerConfig b = current.Broker;
            SecurityConfig s = current.Security;
            BackupsConfig k = current.Backups;
            List<string> applied = new List<string>();
            List<string> restart = new List<string>();
            foreach (KeyValuePair<string, string> u in updates)
            {
                string key = u.Key ?? "";
                string value = u.Value ?? "";
                switch (key)
                {
                    case "security.on_exfil_attempt":
                        s = new SecurityConfig(ParsePolicy(key, value), s.MaxSubstitutionsPerMinute);
                        break;
                    case "security.max_substitutions_per_minute":
                        s = new SecurityConfig(s.OnExfilAttempt, ParseInt(key, value));
                        break;
                    case "backups.max_count":
                        k = new BackupsConfig(ParseInt(key, value));
                        break;
                    case "broker.log_level":
                        {
                            LogLevel level;
                            if (!LogLevelWire.TryParse(value, out level)) throw new ConfigException(key, value);
                            b = new BrokerConfig(b.Listen, b.EventsListen, b.AdminPipe, level, b.EventRetentionDays, b.EventRingSize);
                            break;
                        }
                    case "broker.listen":
                        b = new BrokerConfig(value, b.EventsListen, b.AdminPipe, b.LogLevel, b.EventRetentionDays, b.EventRingSize);
                        break;
                    case "broker.events_listen":
                        b = new BrokerConfig(b.Listen, value, b.AdminPipe, b.LogLevel, b.EventRetentionDays, b.EventRingSize);
                        break;
                    case "broker.admin_pipe":
                        b = new BrokerConfig(b.Listen, b.EventsListen, value, b.LogLevel, b.EventRetentionDays, b.EventRingSize);
                        break;
                    case "broker.event_retention_days":
                        b = new BrokerConfig(b.Listen, b.EventsListen, b.AdminPipe, b.LogLevel, ParseInt(key, value), b.EventRingSize);
                        break;
                    case "broker.event_ring_size":
                        b = new BrokerConfig(b.Listen, b.EventsListen, b.AdminPipe, b.LogLevel, b.EventRetentionDays, ParseInt(key, value));
                        break;
                    default:
                        throw new ConfigKeyException(key);
                }
                if (Array.IndexOf(HotKeys, key) >= 0) applied.Add(key); else restart.Add(key);
            }
            result = new ConfigSetResult(applied, restart);
            return new Config(current.Version, b, s, k, current.Hosts, current.Plugins);
        }

        /// <summary>Champs `broker.*` qui diffèrent (ordre d'IRIS) : ils ne prennent effet qu'au redémarrage (`ignored` de config.reload).</summary>
        public static IList<string> StructuralChanges(Config before, Config after)
        {

        {
            int n;
            if (!int.TryParse(value, NumberStyles.AllowLeadingSign, CultureInfo.InvariantCulture, out n)) throw new ConfigException(key, value);
            return n;
        }

        {
            int index = Array.IndexOf(ExfilPolicyWire.Names, value);
            if (index < 0) throw new ConfigException(key, value);
            return (ExfilPolicy)index;
        }
```

```csharp
        /// <summary>`config.set` : lot atomique (validation par Save avant toute écriture).</summary>
        public ConfigSetResult ApplyUpdates(IList<KeyValuePair<string, string>> updates, DateTime now)
        {
            lock (gate)
            {
                ConfigSetResult result;
                Config next = ConfigUpdates.Apply(Current, updates, out result);
                Save(next, now);
                return result;
            }
        }

        /// <summary>`rule.add` : hôte normalisé et validé ; déjà présent → règle existante, sans écriture.</summary>
        public MITMRule AddHost(string host, DateTime now)
        {
            string h = HostName.Normalize(host);
            if (!HostName.IsValid(h)) throw new ConfigException("hosts.host", host ?? "null");
            lock (gate)
            {
                foreach (MITMRule r in Current.Hosts) if (r.Host == h) return r;
                List<MITMRule> hosts = new List<MITMRule>(Current.Hosts);
                MITMRule rule = new MITMRule(h, now, RuleOrigin.User);
                hosts.Add(rule);
                Save(Current.WithHosts(hosts), now);
                return rule;
            }
        }

        /// <summary>`rule.delete` : false si absent ; hôte `origin: default` → ConfigHostProtectedException.</summary>
        public bool DeleteHost(string host, DateTime now)
        {
            string h = HostName.Normalize(host);
            lock (gate)
            {
                List<MITMRule> hosts = new List<MITMRule>(Current.Hosts);
                int index = hosts.FindIndex(delegate(MITMRule r) { return r.Host == h; });
                if (index < 0) return false;
                if (hosts[index].Origin == RuleOrigin.Default) throw new ConfigHostProtectedException(h);
                hosts.RemoveAt(index);
                Save(Current.WithHosts(hosts), now);
                return true;
            }
        }

        /// <summary>`config.reload` : relit et valide config.json ; en cas d'erreur Current est inchangé (pas de re-seed, contrairement au démarrage). Rend les champs structurels modifiés.</summary>
        public IList<string> Reload()
        {
            lock (gate)
            {
                Config loaded;
                try
                {
                    string text = File.ReadAllText(Path, Encoding.UTF8);
                    loaded = Config.FromJson(Json.ParseObject(text));
                    loaded.Validate();
                }
                catch (Exception ex)
                {
                    if (ex is IOException || ex is UnauthorizedAccessException || ex is JsonException || ex is ConfigException)
                        throw new ConfigReloadException(ex.Message, ex);
                    throw;
                }
                IList<string> ignored = ConfigUpdates.StructuralChanges(Current, loaded);
                Current = loaded;
                return ignored;
            }
```

```csharp
using System;
using System.IO;
using System.Security;

namespace IrisKit
{
    public enum VaultState { Absent, Locked, Unlocked }

    public static class VaultStateWire
    {
        public static string ToWire(VaultState state)
        {
            switch (state)
            {
                case VaultState.Absent: return "absent";
                case VaultState.Locked: return "locked";
                default: return "unlocked";
            }
        }
    }

    public enum VaultGateErrorKind { BadPassword, RateLimited, Exists, Absent }

    /// <summary>Refus de la porte du coffre (spec phase 3 §6) ; message en français, jamais le mot de passe.</summary>
    public sealed class VaultGateException : Exception
    {
        public VaultGateErrorKind Kind { get; private set; }
        public TimeSpan RetryAfter { get; private set; }

        public VaultGateException(VaultGateErrorKind kind, string message, TimeSpan retryAfter) : base(message)
        {
            Kind = kind;
            RetryAfter = retryAfter;
        }
    }

    /// <summary>
    /// Porte du coffre (spec phase 3 §3, §6) : état Absent / Locked / Unlocked. L'ouverture (console ou pipe) charge coffre,
    /// métadonnées, cache et CA, publie le tout d'un coup puis lève Opened. Une ouverture à la fois ; après le n-ième échec,
    /// toute tentative est refusée pendant min(2^(n-1), 60) s sans examiner le mot de passe. Pas de reverrouillage.
    /// </summary>
    public sealed class VaultGate : IDisposable
    {
        public const int MaxDelaySeconds = 60;

        private readonly object gate = new object();
        private readonly object delayGate = new object();
        private readonly string dataDir;
        private readonly string pemPath;
        private readonly Func<KeePassVault, ICAKeyStore> caStoreFor;
        private readonly Func<DateTime> clock;
        private VaultSession session;
        private SecretsMetaStore meta;
        private SecretCache secrets;
        private CAManager ca;
        private volatile bool unlocked;
        private bool inMemory;
        private int failures;
        private DateTime notBefore = DateTime.MinValue;
        private int maxBackups = 10;

        /// <summary>Levé (hors verrou) après chaque ouverture réussie ; un abonné qui lève est journalisé.</summary>
        public event Action Opened;

        public VaultGate(string dataDir, Func<KeePassVault, ICAKeyStore> caStoreFor, Func<DateTime> clock)
        {
            this.dataDir = dataDir;
            this.pemPath = Path.Combine(dataDir, "ca.pem");
            this.caStoreFor = caStoreFor;
            this.clock = clock;
            session = new VaultSession(dataDir);
        }

        /// <summary>Mode `--in-memory-secrets` : porte ouverte d'emblée, secrets en mémoire, CA éphémère (jamais persisté).</summary>
        public static VaultGate InMemory(string dataDir, ISecretStore store, Func<DateTime> clock)
        {
            VaultGate g = new VaultGate(dataDir, null, clock);
            g.inMemory = true;
            g.session = null;
            g.secrets = new SecretCache(store);
            g.ca = new CAManager(new InMemoryCAKeyStore(), g.pemPath, clock);
            g.ca.EnsureCA();
            g.unlocked = true;
            return g;
        }

        public VaultState State
        {
            get
            {
                if (unlocked || inMemory) return VaultState.Unlocked;
                return File.Exists(session.KdbxPath) ? VaultState.Locked : VaultState.Absent;
            }
        }

        public string KdbxPath { get { return inMemory ? null : session.KdbxPath; } }
        public string PemPath { get { return pemPath; } }
        public ISecretStore Secrets { get { return unlocked ? secrets : null; } }
        public CAManager CA { get { return unlocked ? ca : null; } }

        public int MaxBackups
        {
            get { return maxBackups; }
            set
            {
                maxBackups = value;
                KeePassVault v = session == null ? null : session.Vault;
                if (v != null) v.MaxBackups = value;
            }
        }

        /// <summary>Temps restant avant qu'une tentative soit examinée (zéro si aucune attente).</summary>
        public TimeSpan RetryAfter
        {
            get
            {
                lock (delayGate)
                {
                    TimeSpan left = notBefore - clock();
                    return left > TimeSpan.Zero ? left : TimeSpan.Zero;
                }
            }
        }

        /// <summary>Délai imposé après `failures` échecs consécutifs : 1, 2, 4… secondes, plafonné à 60.</summary>
        public static TimeSpan DelayAfter(int failures)
        {
            if (failures <= 0) return TimeSpan.Zero;
            if (failures > 7) return TimeSpan.FromSeconds(MaxDelaySeconds);
            return TimeSpan.FromSeconds(Math.Min(Math.Pow(2, failures - 1), MaxDelaySeconds));
        }

        /// <summary>Déverrouille ; déjà ouvert → retour immédiat sans examen (pas d'oracle). Absent, BadPassword, RateLimited → VaultGateException.</summary>
        public void Unlock(SecureString password)
        {
            lock (gate)
            {
                if (unlocked || inMemory) return;
                if (!session.Exists)
                    throw new VaultGateException(VaultGateErrorKind.Absent, "aucun coffre dans " + session.KdbxPath + " : créez-le d'abord", TimeSpan.Zero);
                CheckDelay();
                try { session.Unlock(password); }
                catch (SecretStoreException ex)
                {
                    if (ex.Kind != SecretStoreErrorKind.VaultLocked) throw;
                    TimeSpan delay = RecordFailure();
                    throw new VaultGateException(VaultGateErrorKind.BadPassword,
                        "mot de passe maître refusé ; nouvel essai possible dans " + (int)Math.Ceiling(delay.TotalSeconds) + " s", delay);
                }
                OpenServices();
                lock (delayGate) { failures = 0; notBefore = DateTime.MinValue; }
            }
            RaiseOpened();
        }

        /// <summary>Crée le coffre (seulement s'il n'existe pas) puis l'ouvre.</summary>
        public void Create(SecureString password)
        {
            lock (gate)
            {
                if (unlocked || inMemory)
                    throw new VaultGateException(VaultGateErrorKind.Exists, "le coffre est déjà ouvert", TimeSpan.Zero);
                if (session.Exists)
                    throw new VaultGateException(VaultGateErrorKind.Exists, "le coffre existe déjà : " + session.KdbxPath, TimeSpan.Zero);
                session.Create(password, clock());
                OpenServices();
            }
            RaiseOpened();
        }

        private void CheckDelay()
        {
            lock (delayGate)
            {
                TimeSpan left = notBefore - clock();
                if (left > TimeSpan.Zero)
                    throw new VaultGateException(VaultGateErrorKind.RateLimited,
                        "trop d'essais : nouvel essai possible dans " + (int)Math.Ceiling(left.TotalSeconds) + " s", left);
            }
        }

        private TimeSpan RecordFailure()
        {
            lock (delayGate)
            {
                failures++;
                TimeSpan delay = DelayAfter(failures);
                notBefore = clock() + delay;
                return delay;
            }
        }

        /// <summary>Sous le verrou : métadonnées, magasin, cache, CA ; en cas d'échec tout est libéré et le coffre refermé.</summary>
        private void OpenServices()
        {
            KeePassVault vault = session.Vault;
            vault.MaxBackups = maxBackups;
            vault.Clock = clock;
            SecretsMetaStore m = null;
            CAManager c = null;
            try
            {
                m = new SecretsMetaStore(Path.Combine(dataDir, "secrets-meta.json"), vault.SecretNames, SecretsMetaStore.DefaultDebounce, clock);
                SecretCache cache = new SecretCache(new KeePassSecretStore(vault, m));
                c = new CAManager(caStoreFor(vault), pemPath, clock);
                c.EnsureCA();
                meta = m;
                secrets = cache;
                ca = c;
                unlocked = true;
                Log.Info("vault", "coffre déverrouillé, MITM actif : " + session.KdbxPath + ", " + vault.SecretNames().Count + " secret(s)");
            }
            catch
            {
                if (c != null) c.Dispose();
                if (m != null) m.Dispose();
                session.Dispose();
                throw;
            }
        }

        private void RaiseOpened()
        {
            Action handler = Opened;
            if (handler == null) return;
            try { handler(); }
            catch (Exception ex) { Log.Error("vault", "abonné à l'ouverture du coffre en échec : " + ex.GetType().Name + " : " + ex.Message); }
        }

        public void Dispose()
        {
            lock (gate)
            {
                unlocked = false;
                if (secrets != null) { secrets.Clear(); secrets = null; }
                if (meta != null) { meta.Dispose(); meta = null; }
                if (ca != null) { ca.Dispose(); ca = null; }
                if (session != null) session.Dispose();
            }
        }
    }
}
```

```csharp
    {
```

```csharp
    public static class AdminPipeSecurity
    {
        public static PipeSecurity Create()
        {
            SecurityIdentifier user = WindowsIdentity.GetCurrent().User;
            PipeSecurity ps = new PipeSecurity();
            ps.SetAccessRuleProtection(true, false);
            ps.SetOwner(user);
            ps.AddAccessRule(new PipeAccessRule(user, PipeAccessRights.FullControl, AccessControlType.Allow));

    /// <summary>Win32 que System.IO.Pipes n'expose pas en .NET Framework 4.8 : FILE_FLAG_FIRST_PIPE_INSTANCE, PIPE_REJECT_REMOTE_CLIENTS, PID du serveur.</summary>
    internal static class PipeNative
    {
        private const uint PIPE_ACCESS_DUPLEX = 0x00000003;
        private const uint FILE_FLAG_FIRST_PIPE_INSTANCE = 0x00080000;
        private const uint FILE_FLAG_OVERLAPPED = 0x40000000;
        private const uint PIPE_TYPE_BYTE = 0x00000000;
        private const uint PIPE_READMODE_BYTE = 0x00000000;
        private const uint PIPE_WAIT = 0x00000000;
        private const uint PIPE_REJECT_REMOTE_CLIENTS = 0x00000008;
        private const int ERROR_ACCESS_DENIED = 5;
        private const uint PROCESS_QUERY_LIMITED_INFORMATION = 0x1000;
        private static readonly IntPtr InvalidHandle = new IntPtr(-1);

        [StructLayout(LayoutKind.Sequential)]
        private struct SECURITY_ATTRIBUTES
        {
            public int nLength;
            public IntPtr lpSecurityDescriptor;
            public int bInheritHandle;
        }

        [DllImport("kernel32.dll", CharSet = CharSet.Unicode, SetLastError = true)]
        private static extern IntPtr CreateNamedPipe(string name, uint openMode, uint pipeMode, uint maxInstances,
            uint outBufferSize, uint inBufferSize, uint defaultTimeout, ref SECURITY_ATTRIBUTES securityAttributes);

        [DllImport("kernel32.dll", SetLastError = true)]
        private static extern bool GetNamedPipeServerProcessId(SafePipeHandle pipe, out uint serverProcessId);

        [DllImport("kernel32.dll", SetLastError = true)]
        private static extern IntPtr OpenProcess(uint access, bool inheritHandle, uint processId);

        [DllImport("kernel32.dll", CharSet = CharSet.Unicode, SetLastError = true)]
        private static extern bool QueryFullProcessImageName(IntPtr process, uint flags, StringBuilder name, ref uint size);

        [DllImport("kernel32.dll", SetLastError = true)]
        private static extern bool CloseHandle(IntPtr handle);

        /// <summary>
        /// Instance serveur asynchrone, mode octet, clients distants refusés, ACL `security`. `first` : échec si le pipe existe déjà
        /// (AdminPipeInUseException). Rend un NamedPipeServerStream non connecté.
        /// </summary>
        public static NamedPipeServerStream CreateServer(string pipeName, bool first, int maxInstances, PipeSecurity security)
        {
            byte[] descriptor = security.GetSecurityDescriptorBinaryForm();
            GCHandle pin = GCHandle.Alloc(descriptor, GCHandleType.Pinned);
            try
            {
                SECURITY_ATTRIBUTES sa = new SECURITY_ATTRIBUTES();
                sa.nLength = Marshal.SizeOf(typeof(SECURITY_ATTRIBUTES));
                sa.lpSecurityDescriptor = pin.AddrOfPinnedObject();
                sa.bInheritHandle = 0;
                uint openMode = PIPE_ACCESS_DUPLEX | FILE_FLAG_OVERLAPPED | (first ? FILE_FLAG_FIRST_PIPE_INSTANCE : 0u);
                uint pipeMode = PIPE_TYPE_BYTE | PIPE_READMODE_BYTE | PIPE_WAIT | PIPE_REJECT_REMOTE_CLIENTS;

                if (raw == InvalidHandle || raw == IntPtr.Zero)
                {
                    int error = Marshal.GetLastWin32Error();
                    if (first && error == ERROR_ACCESS_DENIED) throw new AdminPipeInUseException(pipeName);
                    throw new IOException("création du pipe \\\\.\\pipe\\" + pipeName + " impossible : " + new Win32Exception(error).Message);
                }
                return new NamedPipeServerStream(PipeDirection.InOut, true, false, new SafePipeHandle(raw, true));
            }
            finally { pin.Free(); }
        }

        /// <summary>Chemin complet de l'exécutable du processus serveur d'un pipe client connecté.</summary>
        public static string ServerImagePath(NamedPipeClientStream client)
        {
            uint pid;
            if (!GetNamedPipeServerProcessId(client.SafePipeHandle, out pid))
                throw new IOException("PID du serveur du pipe illisible : " + new Win32Exception(Marshal.GetLastWin32Error()).Message);
            IntPtr process = OpenProcess(PROCESS_QUERY_LIMITED_INFORMATION, false, pid);
            if (process == IntPtr.Zero)
                throw new IOException("processus serveur du pipe inaccessible (PID " + pid + ") : " + new Win32Exception(Marshal.GetLastWin32Error()).Message);
            try
            {
                StringBuilder name = new StringBuilder(32768);
                uint size = (uint)name.Capacity;
                if (!QueryFullProcessImageName(process, 0, name, ref size))
                    throw new IOException("exécutable du serveur du pipe illisible : " + new Win32Exception(Marshal.GetLastWin32Error()).Message);
                return name.ToString(0, (int)size);
            }
            finally { CloseHandle(process); }
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.IO;
using System.IO.Pipes;
using System.Threading;
using System.Threading.Tasks;

namespace IrisKit
{
    /// <summary>
    /// Serveur du pipe admin (spec phase 3 §4) : instances asynchrones (16 au plus), une requête à la fois par connexion,
    /// trame > 1 Mio = fermeture sans réponse, tampon de chaque trame effacé après traitement. Le traitement est délégué
    /// à `handler` (AdminDispatcher.HandleAsync en production), qui ne doit pas lever.
    /// </summary>
    public sealed class AdminServer : IDisposable
    {
        public const int MaxInstances = 16;

        private readonly string pipeName;
        private readonly Func<byte[], Task<byte[]>> handler;
        private readonly PipeSecurity security;
        private readonly SemaphoreSlim slots = new SemaphoreSlim(MaxInstances, MaxInstances);
        private readonly object gate = new object();
        private readonly HashSet<NamedPipeServerStream> open = new HashSet<NamedPipeServerStream>();
        private readonly List<Task> tasks = new List<Task>();
        private CancellationTokenSource stopping;
        private Task acceptLoop;
        private bool stopped;

        public string PipeName { get { return pipeName; } }

        public AdminServer(string pipeName, Func<byte[], Task<byte[]>> handler)
        {
            if (string.IsNullOrEmpty(pipeName)) throw new ArgumentException("nom de pipe vide", "pipeName");
            if (handler == null) throw new ArgumentNullException("handler");
            this.pipeName = pipeName;
            this.handler = handler;
            security = AdminPipeSecurity.Create();
        }

        /// <summary>Crée la première instance (AdminPipeInUseException si le pipe existe déjà) puis lance la boucle d'acceptation.</summary>
        public void Start()
        {
            if (stopping != null) throw new InvalidOperationException("serveur du pipe déjà démarré");
            slots.Wait();
            NamedPipeServerStream first;
            try { first = PipeNative.CreateServer(pipeName, true, MaxInstances, security); }
            catch { slots.Release(); throw; }
            stopping = new CancellationTokenSource();
            acceptLoop = AcceptLoopAsync(first, stopping.Token);
            Log.Info("ipc", "pipe admin à l'écoute : \\\\.\\pipe\\" + pipeName);
        }

        private async Task AcceptLoopAsync(NamedPipeServerStream listening, CancellationToken cancel)
        {
            while (true)
            {
                Track(listening);
                Exception waitError = null;
                try { await Task.Factory.FromAsync(listening.BeginWaitForConnection, listening.EndWaitForConnection, null).ConfigureAwait(false); }
                catch (Exception ex) { waitError = ex; }
                if (cancel.IsCancellationRequested) { Release(listening); return; }
                if (waitError == null)
                {
                    Task t = ServeAsync(listening, cancel);
                    lock (gate) { tasks.Add(t); tasks.RemoveAll(delegate(Task x) { return x.IsCompleted; }); }
                }
                else
                {
                    Release(listening);
                    if (!(waitError is IOException))
                    {
                        Log.Error("ipc", "pipe admin arrêté : " + waitError.GetType().Name + " : " + waitError.Message);
                        return;
                    }
                    // client parti avant l'acceptation : instance jetée, on en crée une autre
                }
                try { await slots.WaitAsync(cancel).ConfigureAwait(false); }
                catch (OperationCanceledException) { return; }
                try { listening = PipeNative.CreateServer(pipeName, false, MaxInstances, security); }
                catch (IOException ex)
                {
                    slots.Release();
                    Log.Error("ipc", "nouvelle instance du pipe admin impossible : " + ex.Message);
                    return;
                }
            }
        }

        private async Task ServeAsync(NamedPipeServerStream pipe, CancellationToken cancel)
        {
            try
            {
                while (!cancel.IsCancellationRequested)
                {
                    byte[] request;
                    try { request = await FrameCodec.ReadAsync(pipe, FrameCodec.MaxRequestBytes, cancel).ConfigureAwait(false); }
                    catch (FrameTooLargeException ex)
                    {
                        Log.Warn("ipc", "trame refusée sur le pipe admin : " + ex.Message + " ; connexion fermée");
                        return;
                    }
                    if (request == null) return;
                    byte[] response;
                    try { response = await handler(request).ConfigureAwait(false); }
                    finally { Array.Clear(request, 0, request.Length); }
                    await FrameCodec.WriteAsync(pipe, response, cancel).ConfigureAwait(false);
                }
            }
            catch (IOException) { }
            catch (ObjectDisposedException) { }
            catch (OperationCanceledException) { }
            catch (Exception ex) { Log.Error("ipc", "erreur inattendue sur le pipe admin : " + ex.GetType().Name + " : " + ex.Message); }
            finally { Release(pipe); }
        }

        private void Track(NamedPipeServerStream pipe)
        {
            lock (gate) { open.Add(pipe); }
        }

        /// <summary>Ferme une instance et rend son créneau (une seule fois par instance suivie).</summary>
        private void Release(NamedPipeServerStream pipe)
        {
            bool removed;
            lock (gate) { removed = open.Remove(pipe); }
            try { pipe.Dispose(); }
            catch (IOException) { }
            if (removed) slots.Release();
        }

        /// <summary>Ferme toutes les instances (connexions en cours abandonnées) et attend la fin des tâches, 2 s au plus.</summary>
        public void Stop()
        {
            if (stopping == null || stopped) return;
            stopped = true;
            stopping.Cancel();
            List<NamedPipeServerStream> snapshot;
            List<Task> pending;
            lock (gate)
            {
                snapshot = new List<NamedPipeServerStream>(open);
                pending = new List<Task>(tasks);
            }
            foreach (NamedPipeServerStream pipe in snapshot)
            {
                try { pipe.Dispose(); }
                catch (IOException) { }
            }
            try { Task.WaitAll(pending.ToArray(), TimeSpan.FromSeconds(2)); }
            catch (AggregateException) { }
            try { acceptLoop.Wait(TimeSpan.FromSeconds(2)); }
            catch (AggregateException) { }
            Log.Info("ipc", "pipe admin fermé");
        }

        public void Dispose() { Stop(); }
    }
}
```

```csharp
using System;
using System.Collections;
using System.IO;
using System.IO.Pipes;
using System.Reflection;
using System.Threading;

namespace IrisKit
{
    /// <summary>Le serveur du pipe n'est pas l'exécutable attendu : rien n'est envoyé (spec phase 3 §4, invariant 4).</summary>
    public sealed class AdminServerIdentityException : Exception
    {
        public string Expected { get; private set; }
        public string Actual { get; private set; }

        public AdminServerIdentityException(string expected, string actual)
            : base("le serveur du pipe n'est pas " + expected + " (exécutable : " + actual + ") : connexion refusée, rien n'a été envoyé")
        {
            Expected = expected;
            Actual = actual;
        }
    }

    /// <summary>
    /// Client du pipe admin (tests, CLI, app) : connexion, vérification de l'identité du serveur avant toute requête,
    /// appels JSON-RPC synchrones, une requête à la fois. Les trames envoyées sont effacées après écriture.
    /// </summary>
    public sealed class AdminClient : IDisposable
    {
        private readonly NamedPipeClientStream pipe;
        private readonly object gate = new object();
        private long nextId;

        private AdminClient(NamedPipeClientStream pipe) { this.pipe = pipe; }

        /// <summary>irisd.exe du dossier de l'exécutable appelant (dossier portable).</summary>
        public static string DefaultServerPath()
        {
            Assembly entry = Assembly.GetEntryAssembly() ?? Assembly.GetExecutingAssembly();
            return Path.Combine(Path.GetDirectoryName(entry.Location), "irisd.exe");
        }

        public static AdminClient Connect(string pipeName, string expectedServerPath, TimeSpan timeout)
        {
            NamedPipeClientStream p = new NamedPipeClientStream(".", pipeName, PipeDirection.InOut, PipeOptions.Asynchronous);
            try
            {
                p.Connect((int)timeout.TotalMilliseconds);
                string actual = PipeNative.ServerImagePath(p);
                if (!string.Equals(Path.GetFullPath(actual), Path.GetFullPath(expectedServerPath), StringComparison.OrdinalIgnoreCase))
                    throw new AdminServerIdentityException(expectedServerPath, actual);
                return new AdminClient(p);
            }
            catch
            {
                p.Dispose();
                throw;
            }
        }

        /// <summary>Appel JSON-RPC : rend `result` (dictionnaire, liste ou scalaire) ou lève RpcException ; IOException si le daemon ferme.</summary>
        public object Call(string method, IDictionary parameters)
        {
            lock (gate)
            {
                long id = ++nextId;
                byte[] request = JsonRpc.Request(id, method, parameters);
                try { FrameCodec.WriteAsync(pipe, request, CancellationToken.None).GetAwaiter().GetResult(); }
                finally { Array.Clear(request, 0, request.Length); }
                byte[] response = FrameCodec.ReadAsync(pipe, FrameCodec.MaxResponseBytes, CancellationToken.None).GetAwaiter().GetResult();
                if (response == null) throw new IOException("le daemon a fermé le pipe sans répondre");
                return JsonRpc.ParseResponse(response, id);
            }
        }

        public void Dispose() { pipe.Dispose(); }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Collections.Specialized;
using System.Globalization;
using System.Threading.Tasks;

namespace IrisKit
{
    /// <summary>Gestionnaire d'une méthode admin : paramètres (null si absents) → résultat JSON (dictionnaire, liste ou scalaire).</summary>
    public delegate Task<object> AdminMethod(JsonObject parameters);

    /// <summary>
    /// Routage JSON-RPC du pipe admin (SPECS §13.2, spec phase 3 §5–§6) : décode, vérifie le coffre pour les méthodes qui l'exigent,
    /// appelle le gestionnaire, traduit toute exception en erreur JSON-RPC. Ne logue que la méthode et l'id, jamais les paramètres.
    /// </summary>
    public sealed class AdminDispatcher
    {
        private readonly Dictionary<string, AdminMethod> methods = new Dictionary<string, AdminMethod>(StringComparer.Ordinal);
        private readonly HashSet<string> needsVault = new HashSet<string>(StringComparer.Ordinal);
        private readonly AdminContext context;

        public AdminDispatcher(AdminContext context)
        {
            if (context == null) throw new ArgumentNullException("context");
            this.context = context;
            SecretMethods.Register(this, context);
            VaultMethods.Register(this, context);
        }

        public ICollection<string> MethodNames { get { return methods.Keys; } }

        /// <summary>Déclare une méthode ; `needsVault` : refusée avec -32020 tant que le coffre est fermé.</summary>
        public void Add(string name, bool needsVault, AdminMethod method)
        {
            methods[name] = method;
            if (needsVault) this.needsVault.Add(name);
        }

        /// <summary>Trame reçue → trame de réponse. Ne lève jamais.</summary>
        public async Task<byte[]> HandleAsync(byte[] payload)
        {
            object id = null;
            object result = null;
            RpcException failure = null;
            try
            {
                RpcRequest request = JsonRpc.ParseRequest(payload, out id);
                Log.Info("ipc", "admin " + request.Method + " id=" + DescribeId(id));
                AdminMethod method;
                if (!methods.TryGetValue(request.Method, out method))
                    throw new RpcException(RpcErrors.MethodNotFound, "Method not found: " + request.Method);
                if (needsVault.Contains(request.Method) && context.Vault.State != VaultState.Unlocked) throw VaultLockedError();
                result = await method(request.Params).ConfigureAwait(false);
            }
            catch (Exception ex) { failure = Map(ex); }
            return failure == null ? JsonRpc.Success(id, result) : JsonRpc.Failure(id, failure);
        }

        public static RpcException VaultLockedError()
        {
            return new RpcException(RpcErrors.VaultLocked, "coffre verrouillé : déverrouillez-le avec vault.unlock");
        }

        /// <summary>Exception → erreur JSON-RPC (codes IRIS et irispc). Les messages ne portent jamais une valeur ni un mot de passe.</summary>
        public static RpcException Map(Exception ex)
        {
            RpcException rpc = ex as RpcException;
            if (rpc != null) return rpc;
            SecretStoreException store = ex as SecretStoreException;
            if (store != null)
            {
                switch (store.Kind)
                {
                    case SecretStoreErrorKind.UnknownSecret: return new RpcException(RpcErrors.UnknownSecret, store.Message);
                    case SecretStoreErrorKind.InvalidName: return new RpcException(RpcErrors.InvalidName, store.Message);
                    case SecretStoreErrorKind.InvalidAllowedHosts: return new RpcException(RpcErrors.InvalidAllowedHosts, store.Message);
                    case SecretStoreErrorKind.Duplicate: return new RpcException(RpcErrors.Duplicate, store.Message);
                    case SecretStoreErrorKind.VaultBusy: return new RpcException(RpcErrors.VaultBusy, store.Message);
                    case SecretStoreErrorKind.VaultLocked: return new RpcException(RpcErrors.VaultLocked, store.Message);
                    default: return new RpcException(RpcErrors.InternalError, store.Message);
                }
            }
            VaultGateException vault = ex as VaultGateException;
            if (vault != null)
            {
                switch (vault.Kind)
                {
                    case VaultGateErrorKind.BadPassword: return new RpcException(RpcErrors.BadPassword, vault.Message);
                    case VaultGateErrorKind.Exists: return new RpcException(RpcErrors.VaultExists, vault.Message);
                    case VaultGateErrorKind.Absent: return new RpcException(RpcErrors.NotFound, vault.Message);
                    default:
                        OrderedDictionary data = new OrderedDictionary();
                        data["retry_after_s"] = (long)Math.Ceiling(vault.RetryAfter.TotalSeconds);
                        return new RpcException(RpcErrors.RateLimited, vault.Message, data);
                }
            }
            ConfigKeyException key = ex as ConfigKeyException;
            if (key != null) return new RpcException(RpcErrors.InvalidParams, "unknown key: " + key.Key);
            ConfigException config = ex as ConfigException;
            if (config != null) return new RpcException(RpcErrors.InvalidParams, "invalid value '" + config.Value + "' for '" + config.Field + "'");
            if (ex is ConfigHostProtectedException) return new RpcException(RpcErrors.RuleProtected, ex.Message);
            if (ex is ConfigReloadException) return new RpcException(RpcErrors.ConfigReloadFailed, "config.reload failed: " + ex.Message);
            if (ex is JsonException || ex is FormatException) return new RpcException(RpcErrors.InvalidParams, "Invalid params: " + ex.Message);
            Log.Error("ipc", "erreur inattendue dans une méthode admin : " + ex.GetType().Name + " : " + ex.Message);
            return new RpcException(RpcErrors.InternalError, ex.Message);
        }

        private static string DescribeId(object id)
        {
            return id == null ? "null" : Convert.ToString(id, CultureInfo.InvariantCulture);
        }
    }

    /// <summary>Lecture des paramètres d'une méthode admin ; tout écart lève JsonException ou RpcException (-32602).</summary>
    internal static class AdminParams
    {
        public static JsonObject Require(JsonObject p)
        {
            if (p == null) throw new RpcException(RpcErrors.InvalidParams, "Method requires params");
            return p;
        }

        public static string String(JsonObject p, string key) { return Require(p).String(key); }

        public static IList<string> Strings(JsonObject p, string key) { return Require(p).StringArray(key); }

        public static bool Bool(JsonObject p, string key)
        {
            object value = Require(p).Raw(key);
            if (!(value is bool)) throw new JsonException("champ " + key + " : booléen attendu");
            return (bool)value;
        }

        /// <summary>Valeur binaire en base64 ; l'appelant efface le tableau après usage.</summary>
        public static byte[] Base64(JsonObject p, string key) { return Convert.FromBase64String(String(p, key)); }
    }
}
```

```csharp
using System.Collections.Specialized;
using System.Security;
using System.Threading.Tasks;

namespace IrisKit
{
    /// <summary>`vault.*` propres à irispc (spec phase 3 §6). Le mot de passe passe en SecureString dès le décodage ; jamais logué.</summary>
    internal static class VaultMethods
    {
        public static void Register(AdminDispatcher d, AdminContext c)
        {
            d.Add("vault.status", false, delegate(JsonObject p)
            {
                OrderedDictionary o = new OrderedDictionary();
                o["state"] = VaultStateWire.ToWire(c.Vault.State);
                o["path"] = c.Vault.KdbxPath;
                return Task.FromResult<object>(o);
            });
            d.Add("vault.unlock", false, delegate(JsonObject p)
            {
                using (SecureString password = Password(p)) c.Vault.Unlock(password);
                return Task.FromResult<object>(Unlocked());
            });
            d.Add("vault.create", false, delegate(JsonObject p)
            {
                using (SecureString password = Password(p)) c.Vault.Create(password);
                return Task.FromResult<object>(Unlocked());
            });
        }

        private static OrderedDictionary Unlocked()
        {
            OrderedDictionary o = new OrderedDictionary();
            o["state"] = "unlocked";
            return o;
        }

        /// <summary>`password` chaîne non vide → SecureString ; la string JSON intermédiaire est une limite du runtime (spec §6).</summary>
        private static SecureString Password(JsonObject p)
        {
            string text = AdminParams.Require(p).Raw("password") as string;
            if (string.IsNullOrEmpty(text)) throw new RpcException(RpcErrors.InvalidParams, "Invalid params: password doit être une chaîne non vide");
            SecureString s = new SecureString();
            foreach (char ch in text) s.AppendChar(ch);
            s.MakeReadOnly();
            return s;
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace IrisKit
{
    /// <summary>Filtres de `events.query` (SPECS §13.2, spec phase 3 §5) : since, until, kind, host, puis les `Limit` plus récents.</summary>
    public sealed class EventQuery
    {
        public const int MaxLimit = 10000;

        public DateTime? Since;
        public DateTime? Until;
        public int Limit = MaxLimit;
        public ISet<EventKind> Kinds;
        public string Host;

        public bool Matches(Event e)
        {
            if (Since.HasValue && e.Timestamp < Since.Value) return false;
            if (Until.HasValue && e.Timestamp > Until.Value) return false;
            if (Kinds != null && !Kinds.Contains(e.Kind)) return false;
            if (Host != null && e.Host != Host) return false;
            return true;
        }

        /// <summary>Ring, complété par le journal JSONL quand `since` précède le plus ancien event du ring ; dédoublonné par id, trié, tronqué.</summary>
        public IList<Event> Run(EventRing ring, JsonlEventStore store, DateTime now)
        {
            IList<Event> live = ring.All;
            Dictionary<Guid, Event> byId = new Dictionary<Guid, Event>();
            foreach (Event e in live) if (Matches(e)) byId[e.Id] = e;
            bool needJournal = Since.HasValue && store != null && (live.Count == 0 || Since.Value < live[0].Timestamp);
            if (needJournal)
            {
                DateTime today = now.ToUniversalTime().Date;
                DateTime first = Since.Value.ToUniversalTime().Date;
                DateTime oldestKept = today.AddDays(-store.RetentionDays);
                if (first < oldestKept) first = oldestKept;
                for (DateTime day = first; day <= today; day = day.AddDays(1))
                    foreach (Event e in store.Read(day))
                        if (Matches(e) && !byId.ContainsKey(e.Id)) byId[e.Id] = e;
            }
            List<Event> all = new List<Event>(byId.Values);
            all.Sort(delegate(Event a, Event b) { return a.Timestamp.CompareTo(b.Timestamp); });
            if (all.Count > Limit) all = all.GetRange(all.Count - Limit, Limit);
            return all;
        }
    }
}
```

```csharp
        {
            X509Store store = new X509Store(StoreName.Root, StoreLocation.CurrentUser);
            try
            {
                store.Open(OpenFlags.ReadOnly | OpenFlags.OpenExistingOnly);

            }
            finally { store.Close(); }
        }
```

```csharp
using System;
using System.Collections.Concurrent;
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;

namespace IrisKit
{
    /// <summary>Filtres du flux SSE (SPECS §14.3) : since (backlog), kind (liste), host (égalité stricte). Paramètre invalide → FormatException (400).</summary>
    public sealed class SseFilters
    {
        public DateTime? Since { get; private set; }
        public ISet<EventKind> Kinds { get; private set; }
        public string Host { get; private set; }

        public static SseFilters Parse(string target)
        {
            SseFilters f = new SseFilters();
            int q = target.IndexOf('?');
            if (q < 0) return f;
            foreach (string pair in target.Substring(q + 1).Split('&'))
            {
                if (pair.Length == 0) continue;
                int eq = pair.IndexOf('=');
                string key = Uri.UnescapeDataString(eq < 0 ? pair : pair.Substring(0, eq));
                string value = eq < 0 ? "" : Uri.UnescapeDataString(pair.Substring(eq + 1));
                switch (key)
                {
                    case "since":
                        try { f.Since = Json.ParseDate(value); }
                        catch (JsonException ex) { throw new FormatException("since invalide : " + ex.Message); }
                        break;
                    case "kind":
                        {
                            HashSet<EventKind> kinds = new HashSet<EventKind>();
                            foreach (string k in value.Split(','))
                            {
                                if (k.Length == 0) continue;
                                try { kinds.Add(EventWire.ParseKind(k)); }
                                catch (JsonException) { throw new FormatException("kind inconnu : " + k); }
                            }
                            f.Kinds = kinds.Count == 0 ? null : kinds;
                            break;
                        }
                    case "host":
                        f.Host = value;
                        break;
                }
            }
            return f;
        }

        public bool Matches(Event e)
        {
            if (Since.HasValue && e.Timestamp < Since.Value) return false;
            if (Kinds != null && !Kinds.Contains(e.Kind)) return false;
            if (Host != null && e.Host != Host) return false;
            return true;
        }

        /// <summary>Trame SSE d'un event (SPECS §14.2) : `event: <kind>`, `id: <uuid>`, `data: <JSON compact>`, ligne vide.</summary>
        public static string Format(Event e)
        {
            return "event: " + EventWire.KindToWire(e.Kind) + "\nid: " + e.Id.ToString("D") + "\ndata: " + Json.EmitCompact(e.ToJson()) + "\n\n";
        }
    }

    /// <summary>File bornée d'un abonné SSE, alimentée par EventsBus sans jamais bloquer l'émetteur ; au débordement, plus rien n'est accepté.</summary>
    internal sealed class SseSubscriber
    {
        private readonly ConcurrentQueue<Event> queue = new ConcurrentQueue<Event>();
        private readonly SemaphoreSlim signal = new SemaphoreSlim(0);
        private readonly int capacity;
        private int count;
        private int rejected;
        private volatile bool overflowed;

        public SseSubscriber(int capacity) { this.capacity = capacity; }

        public bool Overflowed { get { return overflowed; } }

        public void Offer(Event e)
        {
            if (overflowed) { Interlocked.Increment(ref rejected); return; }
            if (Interlocked.Increment(ref count) > capacity)
            {
                Interlocked.Decrement(ref count);
                Interlocked.Increment(ref rejected);
                overflowed = true;
                signal.Release();
                return;
            }
            queue.Enqueue(e);
            signal.Release();
        }

        public bool TryTake(out Event e)
        {
            if (!queue.TryDequeue(out e)) return false;
            Interlocked.Decrement(ref count);
            return true;
        }

        /// <summary>Attend un event ou le débordement ; false à l'expiration (moment du ping).</summary>
        public Task<bool> WaitAsync(TimeSpan timeout, CancellationToken cancel) { return signal.WaitAsync(timeout, cancel); }

        /// <summary>Events non livrés : refusés + encore en file ; vide la file.</summary>
        public int TakeDropped()
        {
            int n = Interlocked.Exchange(ref rejected, 0);
            Event e;
            while (queue.TryDequeue(out e)) n++;
            return n;
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Net;
using System.Net.Sockets;
using System.Text;
using System.Threading;
using System.Threading.Tasks;

namespace IrisKit
{
    /// <summary>
    /// Flux d'events SSE (SPECS §14, spec phase 3 §8) : TcpListener en bouclage, `GET /events`, backlog `since` depuis le ring,
    /// ping périodique, file bornée par abonné (débordement → `event: dropped` puis fermeture), 32 abonnés au plus.
    /// Un abonné qui ne lit plus est coupé après WriteTimeout. Jamais de valeur de secret : les events n'en portent pas.
    /// </summary>
    public sealed class EventsServer : IDisposable
    {
        public const int MaxSubscribers = 32;
        public const int MaxQueued = 1000;
        public const string Role = "flux d'events";

        private readonly EventPipeline events;
        private readonly string host;
        private readonly int port;
        private readonly object gate = new object();
        private readonly HashSet<TcpClient> clients = new HashSet<TcpClient>();
        private readonly List<Task> tasks = new List<Task>();
        private TcpListener listener;
        private CancellationTokenSource stopping;
        private Task acceptLoop;
        private int subscribers;

        public TimeSpan PingInterval = TimeSpan.FromSeconds(15);
        public TimeSpan WriteTimeout = TimeSpan.FromSeconds(10);

        public IPEndPoint ListenEndPoint { get; private set; }
        public int SubscriberCount { get { return Volatile.Read(ref subscribers); } }

        /// <summary>`listen` : `ip:port` (validé par BrokerConfig.ValidateListen ; port 0 accepté pour les tests).</summary>
        public EventsServer(string listen, EventPipeline events)
        {
            if (events == null) throw new ArgumentNullException("events");
            int colon = listen.LastIndexOf(':');
            host = listen.Substring(0, colon);
            port = int.Parse(listen.Substring(colon + 1), CultureInfo.InvariantCulture);
            this.events = events;
        }

        public IPEndPoint Start()
        {
            IPAddress ip;
            if (!IPAddress.TryParse(host, out ip) || !IPAddress.IsLoopback(ip))
                throw new ProxyException("le flux d'events n'écoute que sur une adresse de bouclage (127.0.0.1 ou ::1), pas sur " + host);
            if (listener != null) throw new ProxyException("flux d'events déjà démarré");
            TcpListener l = new TcpListener(ip, port);
            PortBinding.Start(l, 16, Role, port);
            listener = l;
            ListenEndPoint = (IPEndPoint)l.LocalEndpoint;
            stopping = new CancellationTokenSource();
            acceptLoop = AcceptLoopAsync(l, stopping.Token);
            Log.Info("events", "flux SSE à l'écoute : http://" + ListenEndPoint + "/events");
            return ListenEndPoint;
        }

        private async Task AcceptLoopAsync(TcpListener l, CancellationToken cancel)
        {
            while (!cancel.IsCancellationRequested)
            {
                TcpClient client;
                try { client = await l.AcceptTcpClientAsync().ConfigureAwait(false); }
                catch (ObjectDisposedException) { return; }
                catch (InvalidOperationException) { return; }
                catch (SocketException) { if (cancel.IsCancellationRequested) return; continue; }
                client.NoDelay = true;
                Task t = ServeAsync(client, cancel);
                lock (gate) { tasks.Add(t); tasks.RemoveAll(delegate(Task x) { return x.IsCompleted; }); }
            }
        }

        private async Task ServeAsync(TcpClient client, CancellationToken cancel)
        {
            lock (gate) { clients.Add(client); }
            bool counted = false;
            IDisposable subscription = null;
            try
            {
                HttpConnection http = new HttpConnection(client.GetStream());
                http.ReadTimeout = TimeSpan.FromSeconds(10);
                HttpRequestHead head = null;
                bool unreadable = false;
                try { head = await HttpParser.ReadRequestHeadAsync(http, cancel).ConfigureAwait(false); }
                catch (HttpProtocolException) { unreadable = true; }
                if (unreadable) { await ConnectHandler.WriteSimpleResponseAsync(http, 400, "requête illisible", cancel).ConfigureAwait(false); return; }
                if (head == null) return;
                string path = head.Target;
                int q = path.IndexOf('?');
                if (q >= 0) path = path.Substring(0, q);
                if (path != "/events") { await ConnectHandler.WriteSimpleResponseAsync(http, 404, "seul /events est servi", cancel).ConfigureAwait(false); return; }
                if (head.Method != "GET") { await ConnectHandler.WriteSimpleResponseAsync(http, 405, "seul GET est accepté sur /events", cancel).ConfigureAwait(false); return; }
                SseFilters filters = null;
                string invalid = null;
                try { filters = SseFilters.Parse(head.Target); }
                catch (FormatException ex) { invalid = ex.Message; }
                if (filters == null) { await ConnectHandler.WriteSimpleResponseAsync(http, 400, invalid, cancel).ConfigureAwait(false); return; }
                if (Interlocked.Increment(ref subscribers) > MaxSubscribers)
                {
                    Interlocked.Decrement(ref subscribers);
                    await ConnectHandler.WriteSimpleResponseAsync(http, 503, "trop d'abonnés au flux d'events (" + MaxSubscribers + " au plus)", cancel).ConfigureAwait(false);
                    return;
                }
                counted = true;
                SseSubscriber sub = new SseSubscriber(MaxQueued);
                subscription = events.Bus.Subscribe(sub.Offer);
                await StreamAsync(http, sub, filters, cancel).ConfigureAwait(false);
            }
            catch (IOException) { }
            catch (ObjectDisposedException) { }
            catch (TimeoutException ex) { Log.Debug("events", "abonné SSE coupé : " + ex.Message); }
            catch (OperationCanceledException) { }
            catch (Exception ex) { Log.Debug("events", "abonné SSE terminé sur " + ex.GetType().Name + " : " + ex.Message); }
            finally
            {
                if (subscription != null) subscription.Dispose();
                if (counted) Interlocked.Decrement(ref subscribers);
                lock (gate) { clients.Remove(client); }
                client.Close();
            }
        }

        private async Task StreamAsync(HttpConnection http, SseSubscriber sub, SseFilters filters, CancellationToken cancel)
        {
            await WriteTextAsync(http, "HTTP/1.1 200 OK\r\nContent-Type: text/event-stream\r\nCache-Control: no-cache\r\nConnection: close\r\n\r\n: connected\n\n", cancel).ConfigureAwait(false);
            HashSet<Guid> sent = new HashSet<Guid>();
            if (filters.Since.HasValue)
                foreach (Event e in events.Ring.Since(filters.Since.Value))
                    if (filters.Matches(e))
                    {
                        sent.Add(e.Id);
                        await WriteTextAsync(http, SseFilters.Format(e), cancel).ConfigureAwait(false);
                    }
            while (!cancel.IsCancellationRequested)
            {
                bool signaled = await sub.WaitAsync(PingInterval, cancel).ConfigureAwait(false);
                if (!signaled && !sub.Overflowed)
                {
                    await WriteTextAsync(http, ": ping\n\n", cancel).ConfigureAwait(false);
                    continue;
                }
                Event next;
                while (!sub.Overflowed && sub.TryTake(out next))
                {
                    if (sent.Count > 0 && sent.Remove(next.Id)) continue;
                    if (filters.Matches(next)) await WriteTextAsync(http, SseFilters.Format(next), cancel).ConfigureAwait(false);
                }
                if (sub.Overflowed)
                {
                    int dropped = sub.TakeDropped();
                    Log.Warn("events", "abonné SSE en retard de plus de " + MaxQueued + " events : " + dropped + " event(s) non livré(s), connexion fermée");
                    await WriteTextAsync(http, "event: dropped\ndata: {\"count\":" + dropped + "}\n\n", cancel).ConfigureAwait(false);
                    return;
                }
            }
        }

        /// <summary>Écrit puis vide ; un abonné qui ne lit plus depuis WriteTimeout est coupé (TimeoutException).</summary>
        private async Task WriteTextAsync(HttpConnection http, string text, CancellationToken cancel)
        {
            byte[] bytes = Encoding.UTF8.GetBytes(text);
            Task write = WriteAndFlushAsync(http, bytes, cancel);
            Task done = await Task.WhenAny(write, Task.Delay(WriteTimeout)).ConfigureAwait(false);
            if (done != write) throw new TimeoutException("abonné SSE bloqué depuis " + (int)WriteTimeout.TotalSeconds + " s");
            await write.ConfigureAwait(false);
        }

        private static async Task WriteAndFlushAsync(HttpConnection http, byte[] bytes, CancellationToken cancel)
        {
            await http.WriteAsync(bytes, 0, bytes.Length, cancel).ConfigureAwait(false);
            await http.FlushAsync(cancel).ConfigureAwait(false);
        }

        /// <summary>Ferme l'écoute et tous les abonnés ; attend la fin des tâches, 2 s au plus.</summary>
        public void Stop()
        {
            if (listener == null) return;
            stopping.Cancel();
            try { listener.Stop(); }
            catch (SocketException) { }
            List<TcpClient> open;
            List<Task> pending;
            lock (gate)
            {
                open = new List<TcpClient>(clients);
                pending = new List<Task>(tasks);
            }
            foreach (TcpClient c in open) c.Close();
            try { Task.WaitAll(pending.ToArray(), TimeSpan.FromSeconds(2)); }
            catch (AggregateException) { }
            try { acceptLoop.Wait(TimeSpan.FromSeconds(2)); }
            catch (AggregateException) { }
            listener = null;
            Log.Info("events", "flux SSE fermé");
        }

        public void Dispose() { Stop(); }
    }
}
```

```csharp
using System;
using System.IO;
using System.Security;
using System.Threading;
using IrisKit;

namespace Irisd
{
    /// <summary>
    /// Invite console du mot de passe maître (spec socle §4.3, phase 3 §6, §9) : création avec confirmation si le coffre est absent,
    /// sinon déverrouillage ; trois échecs → code 2 ; délai anti-force brute attendu avant chaque essai (compteur partagé avec le pipe).
    /// Un déverrouillage par le pipe pendant la saisie annule l'invite ; Ctrl+C → code 0.
    /// </summary>
    internal sealed class ConsoleUnlock
    {
        private readonly VaultGate vault;
        private readonly IConsoleInput console;
        private readonly TextWriter stderr;

        public ConsoleUnlock(VaultGate vault, IConsoleInput console, TextWriter stderr)
        {
            this.vault = vault;
            this.console = console;
            this.stderr = stderr;
        }

        public int Run(CancellationToken cancel)
        {
            using (CancellationTokenSource prompt = CancellationTokenSource.CreateLinkedTokenSource(cancel))
            {
                Action opened = delegate
                {
                    try { prompt.Cancel(); }
                    catch (ObjectDisposedException) { }
                };
                vault.Opened += opened;
                try
                {
                    if (vault.State == VaultState.Unlocked) return ExitCodes.Ok;
                    return vault.State == VaultState.Absent ? Create(prompt.Token) : Unlock(prompt.Token);
                }
                catch (OperationCanceledException)
                {
                    if (cancel.IsCancellationRequested)
                    {
                        stderr.WriteLine("irisd : arrêt demandé pendant la saisie du mot de passe.");
                        return ExitCodes.Ok;
                    }
                    Log.Info("irisd", "coffre déverrouillé par le pipe admin : invite console abandonnée");
                    return ExitCodes.Ok;
                }
                finally { vault.Opened -= opened; }
            }
        }

        private int Create(CancellationToken token)
        {
            stderr.WriteLine("Aucun coffre dans " + vault.KdbxPath + " : création d'un nouveau coffre KeePass.");
            for (int attempt = 1; attempt <= Daemon.MaxPasswordAttempts; attempt++)
            {
                SecureString first = console.ReadPassword("Mot de passe maître du nouveau coffre : ", token);
                if (first == null) return Closed();
                if (first.Length == 0) { stderr.WriteLine("Le mot de passe ne peut pas être vide (" + attempt + "/" + Daemon.MaxPasswordAttempts + ")."); continue; }
                SecureString second = console.ReadPassword("Confirmez le mot de passe maître : ", token);
                if (second == null) return Closed();
                if (!SecureStrings.Equal(first, second)) { stderr.WriteLine("Les deux saisies ne correspondent pas (" + attempt + "/" + Daemon.MaxPasswordAttempts + ")."); continue; }
                try
                {
                    vault.Create(first);
                    return ExitCodes.Ok;
                }
                catch (VaultGateException ex)
                {
                    if (ex.Kind == VaultGateErrorKind.Exists && vault.State == VaultState.Unlocked) return ExitCodes.Ok; // créé entre-temps par le pipe
                    stderr.WriteLine("irisd : " + ex.Message);
                    return ExitCodes.Vault;
                }
                catch (SecretStoreException ex)
                {
                    stderr.WriteLine("irisd : création du coffre impossible : " + ex.Message);
                    return ExitCodes.Vault;
                }
            }
            stderr.WriteLine("Trois échecs : arrêt sans créer de coffre.");
            return ExitCodes.Vault;
        }

        private int Unlock(CancellationToken token)
        {
            for (int attempt = 1; attempt <= Daemon.MaxPasswordAttempts; attempt++)
            {
                WaitRetry(token);
                SecureString password = console.ReadPassword("Mot de passe maître (" + vault.KdbxPath + ") : ", token);
                if (password == null) return Closed();
                while (true)
                {
                    try
                    {
                        vault.Unlock(password);
                        return ExitCodes.Ok;
                    }
                    catch (VaultGateException ex)
                    {
                        if (ex.Kind == VaultGateErrorKind.RateLimited) { WaitRetry(token); continue; } // délai ouvert par le pipe : la tentative ne compte pas
                        if (ex.Kind == VaultGateErrorKind.BadPassword) { stderr.WriteLine("Mot de passe refusé (" + attempt + "/" + Daemon.MaxPasswordAttempts + ")."); break; }
                        stderr.WriteLine("irisd : " + ex.Message);
                        return ExitCodes.Vault;
                    }
                    catch (SecretStoreException ex)
                    {
                        stderr.WriteLine("irisd : " + ex.Message);
                        return ExitCodes.Vault;
                    }
                }
            }
            stderr.WriteLine("Trois échecs : arrêt.");
            return ExitCodes.Vault;
        }

        /// <summary>Attend la fin du délai anti-force brute (annulable) avant d'examiner une tentative.</summary>
        private void WaitRetry(CancellationToken token)
        {
            TimeSpan left = vault.RetryAfter;
            if (left <= TimeSpan.Zero) return;
            stderr.WriteLine("Nouvel essai possible dans " + (int)Math.Ceiling(left.TotalSeconds) + " s.");
            if (token.WaitHandle.WaitOne(left)) token.ThrowIfCancellationRequested();
        }

        private int Closed()
        {
            stderr.WriteLine("irisd : entrée fermée pendant la saisie du mot de passe, arrêt.");
            return ExitCodes.Vault;
        }
    }
}
```

```csharp
using System;
using System.Diagnostics;
using System.IO;
using IrisKit;

namespace Irisd
{
    /// <summary>
    /// Proxy, flux SSE et pipe admin d'un daemon (spec phase 3 §9), câblés sur la porte du coffre (activation du MITM à l'ouverture)
    /// et sur la config (application à chaud après chaque mutation RPC). Arrêt dans l'ordre pipe → SSE → proxy.
    /// </summary>
    internal sealed class DaemonServices : IDisposable
    {
        private readonly ConfigStore config;
        private readonly VaultGate vault;
        private readonly EventPipeline events;
        private readonly Func<DateTime> clock;

        public ProxyServer Proxy { get; private set; }
        public EventsServer Sse { get; private set; }
        public AdminServer Admin { get; private set; }

        public DaemonServices(ConfigStore config, VaultGate vault, EventPipeline events, Func<DateTime> clock)
        {
            this.config = config;
            this.vault = vault;
            this.events = events;
            this.clock = clock;
        }

        /// <summary>Démarre les trois services ; rend ExitCodes.Ok ou le code de sortie avec un message sur stderr.</summary>
        public int Start(TextWriter stderr)
        {
            Proxy = new ProxyServer(ProxyOptions.FromConfig(config.Current), events, clock);
            vault.Opened += Activate;
            Activate(); // porte déjà ouverte (--in-memory-secrets) ; idempotent
            try { Proxy.Start(); }
            catch (PortBindException ex) { stderr.WriteLine("irisd : " + ex.Message); return ExitCodes.Port; }
            catch (ProxyException ex) { stderr.WriteLine("irisd : " + ex.Message); return ExitCodes.Data; }

            Sse = new EventsServer(config.Current.Broker.EventsListen, events);
            try { Sse.Start(); }
            catch (PortBindException ex) { stderr.WriteLine("irisd : " + ex.Message); return ExitCodes.Port; }
            catch (ProxyException ex) { stderr.WriteLine("irisd : " + ex.Message); return ExitCodes.Data; }

            AdminContext ctx = new AdminContext();
            ctx.Config = config;
            ctx.Vault = vault;
            ctx.Events = events;
            ctx.Daemon = new DaemonControl(Proxy, clock());
            ctx.Clock = clock;
            ctx.OnConfigChanged = ApplyConfig;
            ctx.IsCATrusted = CATrust.IsInCurrentUserRoot;
            AdminDispatcher dispatcher = new AdminDispatcher(ctx);
            Admin = new AdminServer(config.Current.Broker.AdminPipe, dispatcher.HandleAsync);
            try { Admin.Start(); }
            catch (AdminPipeInUseException ex) { stderr.WriteLine("irisd : " + ex.Message); return ExitCodes.Port; }
            catch (IOException ex) { stderr.WriteLine("irisd : pipe admin : " + ex.Message); return ExitCodes.Data; }
            return ExitCodes.Ok;
        }

        private void Activate()
        {
            if (vault.State == VaultState.Unlocked) Proxy.Activate(vault.Secrets, vault.CA);
        }

        /// <summary>Après rule.*, config.set, config.reload : sécurité, whitelist et rotation des backups du coffre, sans redémarrage.</summary>
        private void ApplyConfig(Config c)
        {
            Proxy.UpdateSecurity(c.Security);
            Proxy.UpdateRules(c.Hosts);
            vault.MaxBackups = c.Backups.MaxCount;
        }

        public void Dispose()
        {
            vault.Opened -= Activate;
            if (Admin != null) Admin.Dispose();
            if (Sse != null) Sse.Dispose();
            if (Proxy != null) Proxy.Dispose();
        }
    }

    /// <summary>IDaemonControl du processus courant : pid, démarrage, version, pause du proxy.</summary>
    internal sealed class DaemonControl : IDaemonControl
    {
        private readonly ProxyServer proxy;
        private readonly DateTime startedAt;
        private readonly int processId;

        public DaemonControl(ProxyServer proxy, DateTime startedAt)
        {
            this.proxy = proxy;
            this.startedAt = startedAt;
            processId = Process.GetCurrentProcess().Id;
        }

        public int ProcessId { get { return processId; } }
        public DateTime StartedAt { get { return startedAt; } }
        public string Version { get { return IrisVersion.Value; } }
        public bool IsPaused { get { return proxy.Paused; } }
        public void SetPaused(bool paused) { proxy.SetPaused(paused); }
    }
}
```

```csharp
using System;
using System.IO;
using System.Threading;
using IrisKit;

namespace Irisd
{
    /// <summary>
    /// Séquence de démarrage (spec phase 3 §9) : data\ → config → events → porte du coffre → proxy → SSE → pipe admin →
    /// « irisd ready » → invite console (sauf --locked ou coffre déjà ouvert) → attente de Ctrl+C → arrêt pipe, SSE, proxy,
    /// puis coffre et journal. Rend un code de sortie ; stderr ne reçoit que l'invite et les erreurs fatales, le reste passe par Log.
    /// </summary>
    public sealed class Daemon
    {
        public const int MaxPasswordAttempts = 3;

        private readonly Arguments options;
        private readonly TextWriter stdout;
        private readonly TextWriter stderr;
        private readonly IConsoleInput console;

        public Daemon(Arguments options, TextWriter stdout, TextWriter stderr, IConsoleInput console)
        {
            this.options = options; this.stdout = stdout; this.stderr = stderr; this.console = console;
        }

        public int Run(CancellationToken cancel)
        {
            Func<DateTime> clock = delegate { return DateTime.UtcNow; };
            Log.Writer = stdout;
            LogLevel requested;
            if (options.LogLevel != null && LogLevelWire.TryParse(options.LogLevel, out requested)) Log.Minimum = requested;

            // 1. data\
            string dataDir = DataDir.Resolve(options.Data);
            try { DataDir.Prepare(dataDir); }
            catch (DataDirException ex) { stderr.WriteLine("irisd : " + ex.Message); return ExitCodes.Data; }

            // 2. config.json
            ConfigStore config;
            try { config = ConfigStore.Open(dataDir, clock()); }
            catch (ConfigIOException ex) { stderr.WriteLine("irisd : " + ex.Message); return ExitCodes.Data; }
            if (options.LogLevel == null) Log.Minimum = config.Current.Broker.LogLevel;
            Log.Info("irisd", "irisd " + IrisVersion.Value + ", données : " + dataDir);

            // 3. events
            EventRing ring = new EventRing(config.Current.Broker.EventRingSize);
            JsonlEventStore eventStore = new JsonlEventStore(Path.Combine(dataDir, "events"), config.Current.Broker.EventRetentionDays, clock);
            eventStore.Purge();
            EventPipeline events = new EventPipeline(ring, eventStore, new EventsBus());
            if (config.RecoveredFromCorruption)
                events.Emit(Event.System(clock(), AlertSeverity.High, "config",
                    "config.json corrompu, valeurs par défaut rétablies (copie dans backups\\) : " + config.CorruptionDetail));

            VaultGate vault = null;
            DaemonServices services = null;
            PurgeScheduler purge = null;
            try
            {
                // 4. porte du coffre
                try { vault = OpenGate(dataDir, clock); }
                catch (SecretStoreException ex) { stderr.WriteLine("irisd : CA : " + ex.Message); return ExitCodes.Vault; }
                vault.MaxBackups = config.Current.Backups.MaxCount;
                if (cancel.IsCancellationRequested) { Log.Info("irisd", "arrêt demandé avant le démarrage des services"); return ExitCodes.Ok; }

                // 5. proxy, flux SSE, pipe admin
                services = new DaemonServices(config, vault, events, clock);
                int rc = services.Start(stderr);
                if (rc != ExitCodes.Ok) return rc;
                purge = new PurgeScheduler(eventStore, clock);
                Log.Info("irisd", "irisd ready on " + services.Proxy.ListenEndPoint + ", events " + services.Sse.ListenEndPoint
                    + ", pipe \\\\.\\pipe\\" + config.Current.Broker.AdminPipe + ", coffre " + StateLabel(vault.State)
                    + ", ca=" + vault.PemPath + ". Ctrl+C pour arrêter.");

                // 6. invite console
                if (!options.Locked && vault.State != VaultState.Unlocked)
                {
                    int code = new ConsoleUnlock(vault, console, stderr).Run(cancel);
                    if (code != ExitCodes.Ok) return code;
                }

                // 7. attente
                cancel.WaitHandle.WaitOne();
                Log.Info("irisd", "arrêt demandé");
                return ExitCodes.Ok;
            }
            finally
            {
                if (purge != null) purge.Dispose();
                if (services != null) services.Dispose();
                if (vault != null) vault.Dispose();
                eventStore.Dispose();
            }
        }

        /// <summary>Porte KeePass (CA dans le coffre, ou éphémère avec --in-memory-ca), ou porte en mémoire ouverte d'emblée (--in-memory-secrets).</summary>
        private VaultGate OpenGate(string dataDir, Func<DateTime> clock)
        {
            if (options.InMemorySecrets)
            {
                Log.Warn("secrets", "DEBUG : --in-memory-secrets, aucun coffre ouvert, secrets lus depuis IRIS_SECRET_<NAME> / IRIS_SECRET_<NAME>_HOSTS");
                InMemorySecretStore memory = new InMemorySecretStore();
                int n = EnvSecrets.LoadInto(memory, Environment.GetEnvironmentVariables(), clock());
                Log.Info("secrets", n + " secret(s) chargé(s) depuis l'environnement");
                if (options.InMemoryCa) Log.Warn("ca", "DEBUG : --in-memory-ca, CA régénéré à chaque démarrage, jamais persisté");
                else Log.Warn("ca", "--in-memory-secrets sans coffre ouvert : le CA est aussi éphémère (équivalent à --in-memory-ca)");
                return VaultGate.InMemory(dataDir, memory, clock);
            }
            bool memoryCa = options.InMemoryCa;
            if (memoryCa) Log.Warn("ca", "DEBUG : --in-memory-ca, CA régénéré à chaque démarrage, jamais persisté");
            return new VaultGate(dataDir, delegate(KeePassVault v)
            {
                return memoryCa ? (ICAKeyStore)new InMemoryCAKeyStore() : new KeePassCAKeyStore(v);
            }, clock);
        }

        private static string StateLabel(VaultState state)
        {
            switch (state)
            {
                case VaultState.Absent: return "absent";
                case VaultState.Locked: return "verrouillé";
                default: return "ouvert";
            }
        }
    }
}
```

```powershell
# Demonstration du pipe admin d'irispc : appelle daemon.status et affiche la reponse JSON.
# Aucun secret ne transite ; ce script ne verifie pas l'identite du serveur (la CLI iris le fera).
param([string]$Pipe = "iris-admin")

$client = New-Object System.IO.Pipes.NamedPipeClientStream(".", $Pipe, [System.IO.Pipes.PipeDirection]::InOut)
$client.Connect(5000)
try {
    $json = '{"jsonrpc":"2.0","method":"daemon.status","id":1}'
    $payload = [System.Text.Encoding]::UTF8.GetBytes($json)
    $length = [BitConverter]::GetBytes([uint32]$payload.Length)
    [Array]::Reverse($length)
    $client.Write($length, 0, 4)
    $client.Write($payload, 0, $payload.Length)
    $client.Flush()

    $header = New-Object byte[] 4
    $read = 0
    while ($read -lt 4) { $n = $client.Read($header, $read, 4 - $read); if ($n -eq 0) { throw "pipe ferme" }; $read += $n }
    [Array]::Reverse($header)
    $size = [BitConverter]::ToUInt32($header, 0)
    $body = New-Object byte[] $size
    $read = 0
    while ($read -lt $size) { $n = $client.Read($body, $read, $size - $read); if ($n -eq 0) { throw "pipe ferme" }; $read += $n }
    [System.Text.Encoding]::UTF8.GetString($body)
}
finally { $client.Dispose() }
```
