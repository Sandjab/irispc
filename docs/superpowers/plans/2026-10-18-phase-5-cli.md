# Phase 5 — CLI `iris` : plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `iris.exe`, client du pipe admin, offre les sous-commandes d'IRIS (`secret`, `rule`, `status`, `pause`, `resume`, `config`, `logs`, `ca`, `doctor`, `mcp wrap|unwrap`) plus `vault status|unlock|create`, avec les codes de sortie d'IRIS.

**Architecture:** CLI mince : `src\iris\` ne contient que l'analyse des arguments, le dispatch et le rendu ; `Cli.Run(args, CliContext)` est testé en process contre un vrai daemon (`iris-admin-test-<guid>`). La logique réutilisable par l'app (JSON ordonné et JSONC, patcheur MCP, registre des fichiers enrobés, observateur de fichier, contrôles de `doctor`) vit dans IrisKit. Le client du pipe compare l'**identité de fichier** du serveur à `irisd.exe`.

**Tech Stack:** C# 5, .NET Framework 4.8, `csc.exe` natif, `System.IO.Pipes`, P/Invoke kernel32, `FileSystemWatcher`, `X509Store`, harnais de tests maison (`tests\Check.cs`).

**Spec:** `docs/superpowers/specs/2026-10-18-phase-5-cli-design.md` (s'appuie sur la spec phase 3 `2026-10-18-phase-3-ipc-design.md`, la spec socle et `../Iris/SPECS.md` §4.3, §16, §16.1).

```csharp
        /// <summary>Jonction NTFS (mklink /J, sans droits administrateur) : `link` → `target`.</summary>
        private static void MakeJunction(string link, string target)
        {
            ProcessStartInfo psi = new ProcessStartInfo("cmd.exe", "/c mklink /J \"" + link + "\" \"" + target + "\"");
            psi.UseShellExecute = false;
            psi.CreateNoWindow = true;
            psi.RedirectStandardOutput = true;
            psi.RedirectStandardError = true;
            using (Process p = Process.Start(psi))
            {
                p.StandardOutput.ReadToEnd();
                string err = p.StandardError.ReadToEnd();
                p.WaitForExit();
                if (p.ExitCode != 0) throw new IOException("mklink /J a échoué : " + err);
            }
        }

        {
            try
            {
                if (Directory.Exists(link)) Directory.Delete(link, false);
                return true;
            }
            catch (IOException) { return false; }
            catch (UnauthorizedAccessException) { return false; }
```

```csharp
using System;
using System.ComponentModel;
using System.IO;
using System.Runtime.InteropServices;
using Microsoft.Win32.SafeHandles;

namespace IrisKit
{
    /// <summary>
    /// Identité d'un fichier : numéro de série du volume + index de fichier (GetFileInformationByHandle). Deux chemins qui mènent
    /// au même fichier (subst, jonction, nom court 8.3, casse, lien physique) ont la même identité (spec phase 5 §8).
    /// </summary>
    public sealed class FileIdentity : IEquatable<FileIdentity>
    {
        private const uint FILE_SHARE_ALL = 0x00000007; // lecture | écriture | suppression
        private const uint OPEN_EXISTING = 3;
        private const uint FILE_FLAG_BACKUP_SEMANTICS = 0x02000000;

        [StructLayout(LayoutKind.Sequential)]
        private struct BY_HANDLE_FILE_INFORMATION
        {
            public uint FileAttributes;
            public System.Runtime.InteropServices.ComTypes.FILETIME CreationTime;
            public System.Runtime.InteropServices.ComTypes.FILETIME LastAccessTime;
            public System.Runtime.InteropServices.ComTypes.FILETIME LastWriteTime;
            public uint VolumeSerialNumber;
            public uint FileSizeHigh;
            public uint FileSizeLow;
            public uint NumberOfLinks;
            public uint FileIndexHigh;
            public uint FileIndexLow;
        }

        [DllImport("kernel32.dll", CharSet = CharSet.Unicode, SetLastError = true)]
        private static extern SafeFileHandle CreateFile(string name, uint access, uint share, IntPtr security, uint disposition, uint flags, IntPtr template);

        [DllImport("kernel32.dll", SetLastError = true)]
        private static extern bool GetFileInformationByHandle(SafeFileHandle file, out BY_HANDLE_FILE_INFORMATION info);

        public uint VolumeSerial { get; private set; }
        public ulong FileIndex { get; private set; }

        private FileIdentity(uint volumeSerial, ulong fileIndex)
        {
            VolumeSerial = volumeSerial;
            FileIndex = fileIndex;
        }

        /// <summary>Identité de `path` ; IOException si le fichier ne peut pas être ouvert (absent, chemin invalide, accès refusé).</summary>
        public static FileIdentity Of(string path)
        {
            if (string.IsNullOrEmpty(path)) throw new IOException("chemin vide");
            // Accès 0 et partage total : n'interfère ni avec un exécutable en cours ni avec un fichier ouvert en écriture.
            using (SafeFileHandle h = CreateFile(path, 0, FILE_SHARE_ALL, IntPtr.Zero, OPEN_EXISTING, FILE_FLAG_BACKUP_SEMANTICS, IntPtr.Zero))
            {
                if (h.IsInvalid)
                    throw new IOException("ouverture de " + path + " impossible : " + new Win32Exception(Marshal.GetLastWin32Error()).Message);
                BY_HANDLE_FILE_INFORMATION info;
                if (!GetFileInformationByHandle(h, out info))
                    throw new IOException("identité de " + path + " illisible : " + new Win32Exception(Marshal.GetLastWin32Error()).Message);
                return new FileIdentity(info.VolumeSerialNumber, ((ulong)info.FileIndexHigh << 32) | info.FileIndexLow);
            }
        }

        /// <summary>Vrai si les deux chemins désignent le même fichier ; faux si l'un des deux ne peut pas être ouvert (jamais d'acceptation par défaut).</summary>
        public static bool Same(string a, string b)
        {
            try { return Of(a).Equals(Of(b)); }
            catch (IOException) { return false; }
        }

        public bool Equals(FileIdentity other)
        {
            return other != null && other.VolumeSerial == VolumeSerial && other.FileIndex == FileIndex;
        }

        public override bool Equals(object obj) { return Equals(obj as FileIdentity); }

        public override int GetHashCode() { return VolumeSerial.GetHashCode() ^ FileIndex.GetHashCode(); }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace IrisKit
{
    public enum OrderedJsonKind { Object, Array, String, Integer, Double, Bool, Null }

    /// <summary>
    /// Valeur JSON qui garde l'ordre des clés (port d'OrderedJSONValue d'IRIS) : objet = liste de paires, tableau, chaîne,
    /// entier 64 bits, double, booléen, null. Objets et tableaux sont mutables ; le patcheur travaille sur une copie (Clone).
    /// </summary>
    public sealed class OrderedJsonValue
    {
        public OrderedJsonKind Kind { get; private set; }
        public List<KeyValuePair<string, OrderedJsonValue>> Pairs { get; private set; }
        public List<OrderedJsonValue> Items { get; private set; }
        public string Text { get; private set; }
        public long Integer { get; private set; }
        public double Number { get; private set; }
        public bool Boolean { get; private set; }

        private OrderedJsonValue(OrderedJsonKind kind) { Kind = kind; }

        public bool IsObject { get { return Kind == OrderedJsonKind.Object; } }

        public static OrderedJsonValue NewObject()
        {
            OrderedJsonValue v = new OrderedJsonValue(OrderedJsonKind.Object);
            v.Pairs = new List<KeyValuePair<string, OrderedJsonValue>>();
            return v;
        }

        public static OrderedJsonValue NewArray()
        {
            OrderedJsonValue v = new OrderedJsonValue(OrderedJsonKind.Array);
            v.Items = new List<OrderedJsonValue>();
            return v;
        }

        public static OrderedJsonValue FromString(string text)
        {
            if (text == null) throw new ArgumentNullException("text");
            OrderedJsonValue v = new OrderedJsonValue(OrderedJsonKind.String);
            v.Text = text;
            return v;
        }

        public static OrderedJsonValue FromInteger(long n)
        {
            OrderedJsonValue v = new OrderedJsonValue(OrderedJsonKind.Integer);
            v.Integer = n;
            return v;
        }

        public static OrderedJsonValue FromDouble(double d)
        {
            OrderedJsonValue v = new OrderedJsonValue(OrderedJsonKind.Double);
            v.Number = d;
            return v;
        }

        public static OrderedJsonValue FromBool(bool b)
        {
            OrderedJsonValue v = new OrderedJsonValue(OrderedJsonKind.Bool);
            v.Boolean = b;
            return v;
        }

        public static OrderedJsonValue Null() { return new OrderedJsonValue(OrderedJsonKind.Null); }

        /// <summary>Valeur de `key` dans un objet ; null (C#) si la clé est absente ou si la valeur n'est pas un objet.</summary>
        public OrderedJsonValue Get(string key)
        {
            if (Kind != OrderedJsonKind.Object) return null;
            foreach (KeyValuePair<string, OrderedJsonValue> p in Pairs)
                if (p.Key == key) return p.Value;
            return null;
        }

        /// <summary>Remplace la valeur de `key` à sa place, ou l'ajoute en fin d'objet.</summary>
        public void Set(string key, OrderedJsonValue value)
        {
            if (Kind != OrderedJsonKind.Object) throw new InvalidOperationException("Set sur une valeur JSON qui n'est pas un objet");
            if (value == null) throw new ArgumentNullException("value");
            for (int i = 0; i < Pairs.Count; i++)
            {
                if (Pairs[i].Key != key) continue;
                Pairs[i] = new KeyValuePair<string, OrderedJsonValue>(key, value);
                return;
            }
            Pairs.Add(new KeyValuePair<string, OrderedJsonValue>(key, value));
        }

        /// <summary>Copie profonde des objets et tableaux ; les scalaires sont immuables et partagés.</summary>
        public OrderedJsonValue Clone()
        {
            if (Kind == OrderedJsonKind.Object)
            {
                OrderedJsonValue o = NewObject();
                foreach (KeyValuePair<string, OrderedJsonValue> p in Pairs)
                    o.Pairs.Add(new KeyValuePair<string, OrderedJsonValue>(p.Key, p.Value.Clone()));
                return o;
            }
            if (Kind == OrderedJsonKind.Array)
            {
                OrderedJsonValue a = NewArray();
                foreach (OrderedJsonValue item in Items) a.Items.Add(item.Clone());
                return a;
            }
            return this;
        }
    }

    public enum CommentKind { Line, Block }

    /// <summary>Position (1-indexée) d'un commentaire JSONC : `mcp wrap` refuse d'écrire un fichier qui en contient.</summary>
    public sealed class CommentPosition
    {
        public int Line { get; private set; }
        public int Column { get; private set; }
        public CommentKind Kind { get; private set; }

        public CommentPosition(int line, int column, CommentKind kind)
        {
            Line = line;
            Column = column;
            Kind = kind;
        }
    }

    /// <summary>JSON refusé : message `L<ligne>:<col> : <raison>` en français.</summary>
    public sealed class OrderedJsonException : Exception
    {
        public int Position { get; private set; }
        public int Line { get; private set; }
        public int Column { get; private set; }

        public OrderedJsonException(string reason, int position, int line, int column)
            : base("L" + line + ":" + column + " : " + reason)
        {
            Position = position;
            Line = line;
            Column = column;
        }
    }

    public sealed class OrderedJsonDocument
    {
        public OrderedJsonValue Root { get; private set; }
        public IList<CommentPosition> Comments { get; private set; }

        public OrderedJsonDocument(OrderedJsonValue root, IList<CommentPosition> comments)
        {
            Root = root;
            Comments = comments;
        }
    }
}
```

Créer `src/IrisKit/MCPConfig/OrderedJsonParser.cs` :

```csharp
using System;
using System.Collections.Generic;
using System.Globalization;
using System.Text;

namespace IrisKit
{
    /// <summary>
    /// Parseur JSON ordonné (port d'OrderedJSONDocument.parse d'IRIS) : RFC 8259 strict, ou JSONC (`//`, `/* */`, virgules finales)
    /// avec relevé des positions de commentaires. Clés en double refusées dans les deux modes. Positions en caractères UTF-16.
    /// </summary>
    public static class OrderedJsonParser
    {
        /// <summary>Profondeur maximale : un fichier hostile `[[[[…` ne fait jamais déborder la pile.</summary>
        public const int MaxDepth = 512;

        public static OrderedJsonDocument Parse(string text, bool jsonc)
        {
            if (text == null) throw new ArgumentNullException("text");
            Cursor c = new Cursor(text, jsonc);
            c.SkipWhitespaceAndComments();
            if (c.AtEnd) throw c.Error("document vide", c.Pos);
            OrderedJsonValue root = c.ParseValue(0);
            c.SkipWhitespaceAndComments();
            if (!c.AtEnd) throw c.Error("caractère inattendu après la valeur : " + Show(text[c.Pos]), c.Pos);
            return new OrderedJsonDocument(root, c.Comments.AsReadOnly());
        }

        private static string Show(char c)
        {
            if (c < 0x20) return "U+" + ((int)c).ToString("X4", CultureInfo.InvariantCulture);
            return "'" + c + "'";
        }

        private static bool IsDigit(char c) { return c >= '0' && c <= '9'; }

        private sealed class Cursor
        {
            private readonly string s;
            private readonly bool jsonc;
            public int Pos;
            public readonly List<CommentPosition> Comments = new List<CommentPosition>();
            // Cache de LineCol : le curseur avance, les appels successifs sont O(1) amorti (pas de O(n²) sur un gros fichier commenté).
            private int cachePos;
            private int cacheLine = 1;
            private int cacheColumn = 1;

            public Cursor(string s, bool jsonc)
            {
                this.s = s;
                this.jsonc = jsonc;
            }

            public bool AtEnd { get { return Pos >= s.Length; } }

            public OrderedJsonException Error(string reason, int at)
            {
                int line, column;
                LineCol(at, out line, out column);
                return new OrderedJsonException(reason, at, line, column);
            }

            private void LineCol(int at, out int line, out int column)
            {
                if (at < cachePos)
                {
                    cachePos = 0;
                    cacheLine = 1;
                    cacheColumn = 1;
                }
                for (int i = cachePos; i < at && i < s.Length; i++)
                {
                    if (s[i] == '\n') { cacheLine++; cacheColumn = 1; }
                    else cacheColumn++;
                }
                cachePos = at;
                line = cacheLine;
                column = cacheColumn;
            }

            private CommentPosition CommentAt(int at, CommentKind kind)
            {
                int line, column;
                LineCol(at, out line, out column);
                return new CommentPosition(line, column, kind);
            }

            public void SkipWhitespaceAndComments()
            {
                while (true)
                {
                    while (Pos < s.Length && (s[Pos] == ' ' || s[Pos] == '\t' || s[Pos] == '\n' || s[Pos] == '\r')) Pos++;
                    if (Pos >= s.Length || s[Pos] != '/') return;
                    if (!jsonc) throw Error("commentaire interdit en JSON strict", Pos);
                    if (Pos + 1 >= s.Length) return; // '/' isolé : refusé par l'appelant
                    char next = s[Pos + 1];
                    if (next == '/')
                    {
                        Comments.Add(CommentAt(Pos, CommentKind.Line));
                        Pos += 2;
                        while (Pos < s.Length && s[Pos] != '\n') Pos++;
                        continue;
                    }
                    if (next == '*')
                    {
                        int start = Pos;
                        Comments.Add(CommentAt(start, CommentKind.Block));
                        Pos += 2;
                        bool closed = false;
                        while (Pos + 1 < s.Length)
                        {
                            if (s[Pos] == '*' && s[Pos + 1] == '/')
                            {
                                Pos += 2;
                                closed = true;
                                break;
                            }
                            Pos++;
                        }
                        if (!closed) throw Error("commentaire /* non fermé", start);
                        continue;
                    }
                    return;
                }
            }

            public OrderedJsonValue ParseValue(int depth)
            {
                SkipWhitespaceAndComments();
                if (AtEnd) throw Error("fin de document inattendue", Pos);
                char c = s[Pos];
                switch (c)
                {
                    case '{': return ParseObject(depth + 1);
                    case '[': return ParseArray(depth + 1);
                    case '"': return OrderedJsonValue.FromString(ParseString());
                    case 't': Literal("true"); return OrderedJsonValue.FromBool(true);
                    case 'f': Literal("false"); return OrderedJsonValue.FromBool(false);
                    case 'n': Literal("null"); return OrderedJsonValue.Null();
                    default:
                        if (c == '-' || IsDigit(c)) return ParseNumber();
                        throw Error("caractère inattendu : " + Show(c), Pos);
                }
            }

            private void CheckDepth(int depth)
            {
                if (depth > MaxDepth) throw Error("imbrication trop profonde (plus de " + MaxDepth + " niveaux)", Pos);
            }

            private OrderedJsonValue ParseObject(int depth)
            {
                CheckDepth(depth);
                Pos++; // {
                OrderedJsonValue obj = OrderedJsonValue.NewObject();
                HashSet<string> seen = new HashSet<string>(StringComparer.Ordinal);
                SkipWhitespaceAndComments();
                if (AtEnd) throw Error("fin de document inattendue dans un objet", Pos);
                if (s[Pos] == '}') { Pos++; return obj; }
                while (true)
                {
                    SkipWhitespaceAndComments();
                    if (AtEnd) throw Error("fin de document inattendue dans un objet", Pos);
                    if (s[Pos] != '"') throw Error("clé entre guillemets attendue, trouvé " + Show(s[Pos]), Pos);
                    int keyPos = Pos;
                    string key = ParseString();
                    if (!seen.Add(key)) throw Error("clé en double : \"" + key + "\"", keyPos);
                    SkipWhitespaceAndComments();
                    if (AtEnd || s[Pos] != ':') throw Error("':' attendu après la clé \"" + key + "\"", Pos);
                    Pos++;
                    obj.Pairs.Add(new KeyValuePair<string, OrderedJsonValue>(key, ParseValue(depth)));
                    SkipWhitespaceAndComments();
                    if (AtEnd) throw Error("fin de document inattendue dans un objet", Pos);
                    char next = s[Pos];
                    if (next == '}') { Pos++; return obj; }
                    if (next != ',') throw Error("',' ou '}' attendu, trouvé " + Show(next), Pos);
                    Pos++;
                    SkipWhitespaceAndComments();
                    if (!AtEnd && s[Pos] == '}')
                    {
                        if (!jsonc) throw Error("virgule finale interdite en JSON strict", Pos);
                        Pos++;
                        return obj;
                    }
                }
            }

            private OrderedJsonValue ParseArray(int depth)
            {
                CheckDepth(depth);
                Pos++; // [
                OrderedJsonValue array = OrderedJsonValue.NewArray();
                SkipWhitespaceAndComments();
                if (AtEnd) throw Error("fin de document inattendue dans un tableau", Pos);
                if (s[Pos] == ']') { Pos++; return array; }
                while (true)
                {
                    array.Items.Add(ParseValue(depth));
                    SkipWhitespaceAndComments();
                    if (AtEnd) throw Error("fin de document inattendue dans un tableau", Pos);
                    char next = s[Pos];
                    if (next == ']') { Pos++; return array; }
                    if (next != ',') throw Error("',' ou ']' attendu, trouvé " + Show(next), Pos);
                    Pos++;
                    SkipWhitespaceAndComments();
                    if (!AtEnd && s[Pos] == ']')
                    {
                        if (!jsonc) throw Error("virgule finale interdite en JSON strict", Pos);
                        Pos++;
                        return array;
                    }
                }
            }

            private string ParseString()
            {
                Pos++; // "
                StringBuilder sb = new StringBuilder();
                while (Pos < s.Length)
                {
                    char c = s[Pos];
                    if (c == '"') { Pos++; return sb.ToString(); }
                    if (c == '\\')
                    {
                        int esc = Pos;
                        Pos++;
                        if (AtEnd) throw Error("chaîne non fermée", esc);
                        char e = s[Pos++];
                        switch (e)
                        {
                            case '"': sb.Append('"'); break;
                            case '\\': sb.Append('\\'); break;
                            case '/': sb.Append('/'); break;
                            case 'b': sb.Append('\b'); break;
                            case 'f': sb.Append('\f'); break;
                            case 'n': sb.Append('\n'); break;
                            case 'r': sb.Append('\r'); break;
                            case 't': sb.Append('\t'); break;
                            case 'u': AppendUnicodeEscape(sb, esc); break;
                            default: throw Error("échappement invalide : \\" + e, esc);
                        }
                        continue;
                    }
                    if (c < 0x20) throw Error("caractère de contrôle non échappé dans une chaîne : " + Show(c), Pos);
                    sb.Append(c);
                    Pos++;
                }
                throw Error("chaîne non fermée", Pos);
            }

            private void AppendUnicodeEscape(StringBuilder sb, int esc)
            {
                int high = Hex4(esc);
                if (high >= 0xD800 && high <= 0xDBFF)
                {
                    if (Pos + 1 >= s.Length || s[Pos] != '\\' || s[Pos + 1] != 'u') throw Error("demi-paire de substitution isolée", esc);
                    Pos += 2;
                    int low = Hex4(esc);
                    if (low < 0xDC00 || low > 0xDFFF) throw Error("paire de substitution invalide", esc);
                    sb.Append((char)high);
                    sb.Append((char)low);
                    return;
                }
                if (high >= 0xDC00 && high <= 0xDFFF) throw Error("demi-paire de substitution isolée", esc);
                sb.Append((char)high);
            }

            private int Hex4(int esc)
            {
                if (Pos + 4 > s.Length) throw Error("échappement \\u incomplet", esc);
                int value = 0;
                for (int i = 0; i < 4; i++)
                {
                    char h = s[Pos++];
                    int digit;
                    if (h >= '0' && h <= '9') digit = h - '0';
                    else if (h >= 'a' && h <= 'f') digit = h - 'a' + 10;
                    else if (h >= 'A' && h <= 'F') digit = h - 'A' + 10;
                    else throw Error("échappement \\u invalide", esc);
                    value = value * 16 + digit;
                }
                return value;
            }

            private void Literal(string word)
            {
                if (Pos + word.Length > s.Length || string.CompareOrdinal(s, Pos, word, 0, word.Length) != 0)
                    throw Error("caractère inattendu : " + Show(s[Pos]), Pos);
                Pos += word.Length;
            }

            private OrderedJsonValue ParseNumber()
            {
                int start = Pos;
                bool isFloat = false;
                if (s[Pos] == '-') Pos++;
                if (AtEnd) throw Error("nombre invalide", start);
                if (s[Pos] == '0')
                {
                    Pos++;
                    if (!AtEnd && IsDigit(s[Pos])) throw Error("nombre invalide (zéro en tête)", start);
                }
                else if (s[Pos] >= '1' && s[Pos] <= '9')
                {
                    while (!AtEnd && IsDigit(s[Pos])) Pos++;
                }
                else throw Error("nombre invalide", start);
                if (!AtEnd && s[Pos] == '.')
                {
                    isFloat = true;
                    Pos++;
                    if (AtEnd || !IsDigit(s[Pos])) throw Error("nombre invalide (chiffre attendu après le point)", start);
                    while (!AtEnd && IsDigit(s[Pos])) Pos++;
                }
                if (!AtEnd && (s[Pos] == 'e' || s[Pos] == 'E'))
                {
                    isFloat = true;
                    Pos++;
                    if (!AtEnd && (s[Pos] == '+' || s[Pos] == '-')) Pos++;
                    if (AtEnd || !IsDigit(s[Pos])) throw Error("nombre invalide (exposant)", start);
                    while (!AtEnd && IsDigit(s[Pos])) Pos++;
                }
                string text = s.Substring(start, Pos - start);
                long n;
                if (!isFloat && long.TryParse(text, NumberStyles.AllowLeadingSign, CultureInfo.InvariantCulture, out n))
                    return OrderedJsonValue.FromInteger(n);
                double d;
                if (double.TryParse(text, NumberStyles.Float, CultureInfo.InvariantCulture, out d) && !double.IsInfinity(d) && !double.IsNaN(d))
                    return OrderedJsonValue.FromDouble(d);
                throw Error("nombre hors limites : " + text, start);
            }
        }
    }
}
```

```csharp
using System;
using System.Globalization;
using System.Text;

namespace IrisKit
{
    /// <summary>Sérialiseur d'OrderedJsonValue (OrderedJSONDocument.serialize d'IRIS) : 2 espaces, `": "`, LF, saut de ligne final.</summary>
    public static class OrderedJsonWriter
    {
        public static string Serialize(OrderedJsonValue root)
        {
            StringBuilder sb = new StringBuilder();
            Emit(sb, root, 0);
            sb.Append('\n');
            return sb.ToString();
        }

        /// <summary>Chaîne JSON entre guillemets (mêmes échappements que la sérialisation) ; sert aussi au diff de --dry-run.</summary>
        public static string Quote(string text)
        {
            StringBuilder sb = new StringBuilder();
            AppendString(sb, text);
            return sb.ToString();
        }

        private static void Emit(StringBuilder sb, OrderedJsonValue v, int indent)
        {
            switch (v.Kind)
            {
                case OrderedJsonKind.Null: sb.Append("null"); return;
                case OrderedJsonKind.Bool: sb.Append(v.Boolean ? "true" : "false"); return;
                case OrderedJsonKind.Integer: sb.Append(v.Integer.ToString(CultureInfo.InvariantCulture)); return;
                case OrderedJsonKind.Double: sb.Append(FormatDouble(v.Number)); return;
                case OrderedJsonKind.String: AppendString(sb, v.Text); return;
                case OrderedJsonKind.Array:
                    if (v.Items.Count == 0) { sb.Append("[]"); return; }
                    sb.Append("[\n");
                    for (int i = 0; i < v.Items.Count; i++)
                    {
                        Pad(sb, indent + 1);
                        Emit(sb, v.Items[i], indent + 1);
                        if (i < v.Items.Count - 1) sb.Append(',');
                        sb.Append('\n');
                    }
                    Pad(sb, indent);
                    sb.Append(']');
                    return;
                default:
                    if (v.Pairs.Count == 0) { sb.Append("{}"); return; }
                    sb.Append("{\n");
                    for (int i = 0; i < v.Pairs.Count; i++)
                    {
                        Pad(sb, indent + 1);
                        AppendString(sb, v.Pairs[i].Key);
                        sb.Append(": ");
                        Emit(sb, v.Pairs[i].Value, indent + 1);
                        if (i < v.Pairs.Count - 1) sb.Append(',');
                        sb.Append('\n');
                    }
                    Pad(sb, indent);
                    sb.Append('}');
                    return;
            }
        }

        private static void Pad(StringBuilder sb, int indent) { sb.Append(' ', indent * 2); }

        /// <summary>Double entier (|d| &lt; 1e15) : forme fractionnaire `1.0` comme IRIS ; sinon forme aller-retour (« R »).</summary>
        private static string FormatDouble(double d)
        {
            if (d == Math.Floor(d) && Math.Abs(d) < 1e15) return d.ToString("0.0", CultureInfo.InvariantCulture);
            return d.ToString("R", CultureInfo.InvariantCulture);
        }

        private static void AppendString(StringBuilder sb, string s)
        {
            sb.Append('"');
            foreach (char c in s)
            {
                switch (c)
                {
                    case '"': sb.Append("\\\""); break;
                    case '\\': sb.Append("\\\\"); break;
                    case '\b': sb.Append("\\b"); break;
                    case '\f': sb.Append("\\f"); break;
                    case '\n': sb.Append("\\n"); break;
                    case '\r': sb.Append("\\r"); break;
                    case '\t': sb.Append("\\t"); break;
                    default:
                        if (c < 0x20) sb.Append("\\u").Append(((int)c).ToString("x4", CultureInfo.InvariantCulture));
                        else sb.Append(c);
                        break;
                }
            }
            sb.Append('"');
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace IrisKit
{
    public sealed class McpPatchSummary
    {
        public int Patched;
        public int AlreadyCompliant;
        public int Skipped;
    }

    /// <summary>Résultat d'un patch : copie patchée (l'entrée n'est jamais modifiée), compteurs, ajouts lisibles pour --dry-run.</summary>
    public sealed class McpPatchResult
    {
        public OrderedJsonValue Root { get; internal set; }
        public McpPatchSummary Summary { get; internal set; }
        /// <summary>Une ligne par clé ajoutée : `mcpServers.github.env.HTTPS_PROXY = "http://127.0.0.1:8888"`.</summary>
        public IList<string> Additions { get; internal set; }
        /// <summary>Serveurs `type` http ou sse sans `env`, laissés tels quels (règle 3).</summary>
        public IList<string> SkippedServers { get; internal set; }
    }

    /// <summary>
    /// `iris mcp wrap` sans E/S (SPECS §16.1, spec phase 5 §6.1) : pour chaque serveur de `mcpServers` (à la racine, sinon dans le
    /// premier objet de premier niveau qui le contient), ajoute à `env` les six variables proxy/CA absentes, sans jamais écraser une
    /// valeur existante. Écart assumé avec le code Swift : HTTPS_PROXY et HTTP_PROXY ne sont pas traités en bloc (SPECS prime).
    /// </summary>
    public static class McpPatcher
    {
        public static readonly string[] ProxyKeys = { "HTTPS_PROXY", "HTTP_PROXY" };
        public static readonly string[] CaKeys = { "CURL_CA_BUNDLE", "NODE_EXTRA_CA_CERTS", "REQUESTS_CA_BUNDLE", "SSL_CERT_FILE" };

        public static McpPatchResult Patch(OrderedJsonValue root, string brokerListen, string caPemPath)
        {
            McpPatchResult r = new McpPatchResult();
            r.Root = root.Clone();
            r.Summary = new McpPatchSummary();
            r.Additions = new List<string>();
            r.SkippedServers = new List<string>();
            string prefix;
            OrderedJsonValue servers = Locate(r.Root, out prefix);
            if (servers == null || !servers.IsObject) return r;
            List<KeyValuePair<string, string>> needed = Needed(brokerListen, caPemPath);
            foreach (KeyValuePair<string, OrderedJsonValue> entry in servers.Pairs)
            {
                OrderedJsonValue server = entry.Value;
                if (!server.IsObject) continue;
                OrderedJsonValue type = server.Get("type");
                bool remote = type != null && type.Kind == OrderedJsonKind.String && (type.Text == "http" || type.Text == "sse");
                OrderedJsonValue env = server.Get("env");
                if (remote && env == null)
                {
                    r.Summary.Skipped++;
                    r.SkippedServers.Add(entry.Key);
                    continue;
                }
                if (env == null || !env.IsObject)
                {
                    // Comme IRIS : `env` absent est créé ; un `env` qui n'est pas un objet ne contient rien d'utilisable et est remplacé.
                    env = OrderedJsonValue.NewObject();
                    server.Set("env", env);
                }
                bool added = false;
                foreach (KeyValuePair<string, string> kv in needed)
                {
                    if (env.Get(kv.Key) != null) continue;
                    env.Set(kv.Key, OrderedJsonValue.FromString(kv.Value));
                    r.Additions.Add(prefix + "." + entry.Key + ".env." + kv.Key + " = " + OrderedJsonWriter.Quote(kv.Value));
                    added = true;
                }
                if (added) r.Summary.Patched++;
                else r.Summary.AlreadyCompliant++;
            }
            return r;
        }

        /// <summary>Les six variables, triées par ordre ordinal (déterminisme, comme IRIS).</summary>
        private static List<KeyValuePair<string, string>> Needed(string brokerListen, string caPemPath)
        {
            List<KeyValuePair<string, string>> needed = new List<KeyValuePair<string, string>>();
            foreach (string k in ProxyKeys) needed.Add(new KeyValuePair<string, string>(k, "http://" + brokerListen));
            foreach (string k in CaKeys) needed.Add(new KeyValuePair<string, string>(k, caPemPath));
            needed.Sort(delegate(KeyValuePair<string, string> a, KeyValuePair<string, string> b) { return string.CompareOrdinal(a.Key, b.Key); });
            return needed;
        }

        private static OrderedJsonValue Locate(OrderedJsonValue root, out string prefix)
        {
            prefix = "mcpServers";
            if (!root.IsObject) return null;
            OrderedJsonValue direct = root.Get("mcpServers");
            if (direct != null) return direct;
            foreach (KeyValuePair<string, OrderedJsonValue> p in root.Pairs)
            {
                if (!p.Value.IsObject) continue;
                OrderedJsonValue nested = p.Value.Get("mcpServers");
                if (nested == null) continue;
                prefix = p.Key + ".mcpServers";
                return nested;
            }
            return null;
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.IO;
using System.Text;

namespace IrisKit
{
    /// <summary>Manifeste présent mais illisible : jamais traité comme vide (un Add le réécrirait et perdrait les autres chemins).</summary>
    public sealed class WrappedPathsCorruptException : Exception
    {
        public WrappedPathsCorruptException(string path, string reason)
            : base("registre des fichiers enrobés illisible : " + path + " (" + reason + ") ; corrigez-le ou supprimez-le") { }
    }

    /// <summary>
    /// Chemins absolus des fichiers MCP patchés par `iris mcp wrap` (`data\wrapped-paths.json`, tableau JSON, sans doublon, ordre
    /// d'insertion) : l'app (phase 6) et la désinstallation (phase 7) les restaurent. Comparaison insensible à la casse (Windows).
    /// </summary>
    public sealed class WrappedPathsRegistry
    {
        private static readonly UTF8Encoding Utf8NoBom = new UTF8Encoding(false);

        public string ManifestPath { get; private set; }

        public WrappedPathsRegistry(string manifestPath) { ManifestPath = manifestPath; }

        public static WrappedPathsRegistry ForDataDir(string dataDir)
        {
            return new WrappedPathsRegistry(Path.Combine(dataDir, "wrapped-paths.json"));
        }

        public IList<string> List()
        {
            if (!File.Exists(ManifestPath)) return new List<string>();
            string text;
            try { text = File.ReadAllText(ManifestPath, Encoding.UTF8); }
            catch (IOException ex) { throw new WrappedPathsCorruptException(ManifestPath, ex.Message); }
            catch (UnauthorizedAccessException ex) { throw new WrappedPathsCorruptException(ManifestPath, ex.Message); }
            IList<object> items;
            try { items = Json.Parse(text) as IList<object>; }
            catch (JsonException ex) { throw new WrappedPathsCorruptException(ManifestPath, ex.Message); }
            if (items == null) throw new WrappedPathsCorruptException(ManifestPath, "tableau JSON attendu");
            List<string> paths = new List<string>();
            foreach (object item in items)
            {
                string s = item as string;
                if (s == null) throw new WrappedPathsCorruptException(ManifestPath, "chemins (chaînes) attendus");
                paths.Add(s);
            }
            return paths;
        }

        public void Add(string path)
        {
            IList<string> paths = List();
            foreach (string p in paths)
                if (string.Equals(p, path, StringComparison.OrdinalIgnoreCase)) return;
            paths.Add(path);
            Write(paths);
        }

        public void Remove(string path)
        {
            IList<string> paths = List();
            List<string> kept = new List<string>();
            foreach (string p in paths)
                if (!string.Equals(p, path, StringComparison.OrdinalIgnoreCase)) kept.Add(p);
            if (kept.Count != paths.Count) Write(kept);
        }

        private void Write(IList<string> paths)
        {
            Directory.CreateDirectory(Path.GetDirectoryName(ManifestPath));
            List<object> items = new List<object>();
            foreach (string p in paths) items.Add(p);
            string tmp = ManifestPath + ".tmp";
            File.WriteAllText(tmp, Json.Emit(items) + "\n", Utf8NoBom);
            if (File.Exists(ManifestPath)) File.Replace(tmp, ManifestPath, null);
            else File.Move(tmp, ManifestPath);
        }
    }
}
```

```csharp
using System;
using System.Security;
using System.Threading;
using System.Threading.Tasks;

namespace IrisKit
{
    /// <summary>Saisie d'un mot de passe ou d'une valeur sans écho. null = entrée fermée (EOF) ; annulation = OperationCanceledException. Jamais loguée.</summary>
    public interface IPasswordReader
    {
        SecureString ReadPassword(string prompt, CancellationToken cancel);
    }

    /// <summary>
    /// Console réelle (irisd et iris) : saisie masquée (ReadKey) en surveillant l'annulation entre deux touches ; si l'entrée est
    /// redirigée, ReadLine avec avertissement unique (irisd seulement : iris refuse avant, spec phase 5 §5).
    /// </summary>
    public sealed class ConsolePasswordReader : IPasswordReader
    {
        private bool warnedRedirected;

        public SecureString ReadPassword(string prompt, CancellationToken cancel)
        {
            Console.Error.Write(prompt);
            SecureString s = new SecureString();
            if (Console.IsInputRedirected)
            {
                if (!warnedRedirected)
                {
                    Console.Error.WriteLine();
                    Console.Error.WriteLine("(entrée redirigée : la saisie ne peut pas être masquée)");
                    warnedRedirected = true;
                }
                Task<string> line = Console.In.ReadLineAsync();
                int which = WaitHandle.WaitAny(new[] { ((IAsyncResult)line).AsyncWaitHandle, cancel.WaitHandle });
                if (which == 1) { Console.Error.WriteLine(); throw new OperationCanceledException(cancel); }
                string text;
                try { text = line.GetAwaiter().GetResult(); }
                catch (System.IO.IOException) { return null; }
                if (text == null) return null;
                foreach (char c in text) s.AppendChar(c);
                s.MakeReadOnly();
                return s;
            }
            while (true)
            {
                while (!Console.KeyAvailable)
                {
                    if (cancel.WaitHandle.WaitOne(50)) { Console.Error.WriteLine(); throw new OperationCanceledException(cancel); }
                }
                ConsoleKeyInfo key;
                try { key = Console.ReadKey(true); }
                catch (InvalidOperationException) { return null; }
                if (key.Key == ConsoleKey.Enter) break;
                if (key.Key == ConsoleKey.Backspace) { if (s.Length > 0) s.RemoveAt(s.Length - 1); continue; }
                if (key.KeyChar != '\0') s.AppendChar(key.KeyChar);
            }
            Console.Error.WriteLine();
            s.MakeReadOnly();
            return s;
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace IrisCli
{
    /// <summary>Codes de sortie d'IRIS (SPECS §16, spec phase 5 §3).</summary>
    public static class CliExit
    {
        public const int Ok = 0;
        public const int Logic = 1;
        public const int Unreachable = 2;
        public const int Io = 3;
        public const int Usage = 64;
    }

    /// <summary>Ligne de commande invalide : code 64, message en français suivi de `voir : iris … --help`.</summary>
    public sealed class UsageException : Exception
    {
        public UsageException(string message) : base(message) { }
    }

    /// <summary>Échec d'une commande avec son code de sortie (1 logique, 3 E/S locale, 64 usage métier). Jamais de valeur de secret dans le message.</summary>
    public sealed class CliException : Exception
    {
        public int Code { get; private set; }

        public CliException(int code, string message) : base(message) { Code = code; }
    }

    public delegate int CommandHandler(ParsedArgs args, CliContext ctx);

    /// <summary>Une sous-commande : chemin (« secret add »), aide, positionnels, options à valeur, drapeaux, gestionnaire.</summary>
    public sealed class CommandSpec
    {
        public string Path;
        /// <summary>Sans le préfixe « iris », par exemple « secret add NAME [--allowed-hosts h1,h2] [--json] ».</summary>
        public string Usage;
        public string Summary;
        public string[] Positionals = new string[0];
        public string[] Options = new string[0];
        public string[] Flags = new string[0];
        /// <summary>Parle au daemon : accepte en plus --pipe-name et --config-path.</summary>
        public bool UsesDaemon;
        public CommandHandler Run;
    }

    public sealed class ParsedArgs
    {
        private readonly List<string> positionals = new List<string>();
        private readonly Dictionary<string, string> options = new Dictionary<string, string>(StringComparer.Ordinal);
        private readonly HashSet<string> flags = new HashSet<string>(StringComparer.Ordinal);

        public bool HelpRequested { get; internal set; }

        public string Positional(int index) { return positionals[index]; }

        public string Option(string name)
        {
            string value;
            return options.TryGetValue(name, out value) ? value : null;
        }

        public bool Flag(string name) { return flags.Contains(name); }

        public bool Json { get { return Flag("--json"); } }

        internal int PositionalCount { get { return positionals.Count; } }
        internal void AddPositional(string value) { positionals.Add(value); }
        internal bool HasOption(string name) { return options.ContainsKey(name); }
        internal void SetOption(string name, string value) { options[name] = value; }
        internal void SetFlag(string name) { flags.Add(name); }
    }

    /// <summary>
    /// Analyseur strict (comme celui d'irisd) : options longues exactes et sensibles à la casse, `--nom valeur` ou drapeau `--nom`,
    /// positionnels ; `-h` seule forme courte. Inconnu, répété, valeur manquante ou vide, positionnel manquant ou en trop → UsageException.
    /// </summary>
    public static class CliArguments
    {
        public static ParsedArgs Parse(CommandSpec spec, string[] args, int start)
        {
            ParsedArgs p = new ParsedArgs();
            List<string> valueOptions = new List<string>(spec.Options);
            if (spec.UsesDaemon)
            {
                valueOptions.Add("--pipe-name");
                valueOptions.Add("--config-path");
            }
            for (int i = start; i < args.Length; i++)
            {
                string a = args[i];
                if (a == "--help" || a == "-h")
                {
                    p.HelpRequested = true;
                    return p;
                }
                if (a.StartsWith("--", StringComparison.Ordinal))
                {
                    if (valueOptions.Contains(a))
                    {
                        if (p.HasOption(a)) throw new UsageException("option répétée : " + a);
                        if (i + 1 >= args.Length || args[i + 1].StartsWith("--", StringComparison.Ordinal)) throw new UsageException("valeur manquante pour " + a);
                        string value = args[++i];
                        if (value.Trim().Length == 0) throw new UsageException("valeur vide pour " + a);
                        p.SetOption(a, value);
                    }
                    else if (Array.IndexOf(spec.Flags, a) >= 0)
                    {
                        if (p.Flag(a)) throw new UsageException("option répétée : " + a);
                        p.SetFlag(a);
                    }
                    else throw new UsageException("option inconnue : " + a);
                    continue;
                }
                if (p.PositionalCount >= spec.Positionals.Length) throw new UsageException("argument en trop : " + a);
                p.AddPositional(a);
            }
            if (p.PositionalCount < spec.Positionals.Length) throw new UsageException("argument manquant : " + spec.Positionals[p.PositionalCount]);
            return p;
        }
    }
```

```csharp
    /// <summary>Daemon injoignable (pipe absent, délai dépassé, accès refusé, connexion perdue) : code 2.</summary>
    public sealed class DaemonUnreachableException : Exception
    {
        public DaemonUnreachableException(string message) : base(message) { }

        /// <summary>Message canonique : comment lancer irisd, puis le pipe visé (spec phase 5 §3).</summary>
        public static DaemonUnreachableException For(string pipeName, CliContext ctx, string detail)
        {
            string message = "irisd ne tourne pas. Lancez : \"" + ctx.ExpectedServerPath + "\" --foreground"
                + System.Environment.NewLine + "  pipe : \\\\.\\pipe\\" + pipeName;
            if (detail != null) message += System.Environment.NewLine + "  détail : " + detail;
            return new DaemonUnreachableException(message);
        }

    public static class Paths
    {
        /// <summary>`~` ou `~\…` en tête → profil utilisateur, puis chemin absolu (ArgumentException / NotSupportedException si invalide).</summary>
        public static string ExpandHome(string path, string userProfile)
        {
            if (path == "~") path = userProfile;
            else if (path.StartsWith("~\\", StringComparison.Ordinal) || path.StartsWith("~/", StringComparison.Ordinal))
                path = Path.Combine(userProfile, path.Substring(2));
            return Path.GetFullPath(path);
        }
    }

    public static class PipeNames
    {
        public const string Default = "iris-admin";

        /// <summary>--pipe-name > --config-path > data\config.json voisin de iris.exe > iris-admin (spec phase 5 §3).</summary>
        public static string Resolve(string pipeOption, string configPathOption, string dataDirectory, string userProfile)
        {
            if (pipeOption != null) return pipeOption;
            if (configPathOption != null)
            {
                string path = configPathOption;
                try
                {
                    path = Paths.ExpandHome(configPathOption, userProfile);
                    return FromConfig(path);
                }
                catch (Exception ex)
                {
                    if (ex is IOException || ex is UnauthorizedAccessException || ex is JsonException || ex is ArgumentException || ex is NotSupportedException)
                        throw new CliException(CliExit.Logic, "config illisible : " + path + " (" + ex.Message + ")");
                    throw;
                }
            }
            string local = Path.Combine(dataDirectory, "config.json");
            try
            {
                if (File.Exists(local)) return FromConfig(local);
            }
            catch (IOException) { }
            catch (UnauthorizedAccessException) { }
            catch (JsonException) { }
            return Default;
        }

        private static string FromConfig(string path)
        {
            JsonObject o = Json.ParseObject(File.ReadAllText(path, Encoding.UTF8));
            string name = o.Object("broker").String("admin_pipe");
            if (name.Trim().Length == 0) throw new JsonException("broker.admin_pipe vide");
            return name;
        }
    }

    /// <summary>Connexion au pipe admin pour une commande : identité du serveur vérifiée, erreurs de transport → code 2.</summary>

    {
        private readonly AdminClient client;

        public string PipeName { get; private set; }

        private CliConnection(AdminClient client, string pipeName)
        {
            this.client = client;
            PipeName = pipeName;
        }

        public static string ResolvePipe(ParsedArgs args, CliContext ctx)
        {
            return PipeNames.Resolve(args.Option("--pipe-name"), args.Option("--config-path"), ctx.DataDirectory, ctx.UserProfile);
        }

        public static CliConnection Open(ParsedArgs args, CliContext ctx) { return Open(ResolvePipe(args, ctx), ctx); }

        /// <summary>AdminServerIdentityException traverse (code 2, message dédié) ; pipe absent, délai, accès refusé → DaemonUnreachableException.</summary>
        public static CliConnection Open(string pipeName, CliContext ctx)
        {
            try { return new CliConnection(AdminClient.Connect(pipeName, ctx.ExpectedServerPath, ctx.ConnectTimeout), pipeName); }
            catch (TimeoutException) { throw DaemonUnreachableException.For(pipeName, ctx, null); }
            catch (IOException ex) { throw DaemonUnreachableException.For(pipeName, ctx, ex.Message); }
            catch (UnauthorizedAccessException ex) { throw DaemonUnreachableException.For(pipeName, ctx, ex.Message); }
        }

        /// <summary>Résultat brut (dictionnaire, liste ou scalaire) ; pipe fermé en cours d'appel → DaemonUnreachableException.</summary>
        public object Call(string method, IDictionary parameters)
        {
            try { return client.Call(method, parameters); }
            catch (IOException ex) { throw new DaemonUnreachableException("connexion au daemon perdue (\\\\.\\pipe\\" + PipeName + ") : " + ex.Message); }
        }

        public static JsonObject AsObject(object raw)
        {
            IDictionary<string, object> d = raw as IDictionary<string, object>;
            if (d == null) throw new CliException(CliExit.Logic, "réponse inattendue du daemon (objet attendu)");
            return new JsonObject(d);
        }

        public static IList<object> AsList(object raw)
        {
            IList<object> list = raw as IList<object>;
            if (list == null) throw new CliException(CliExit.Logic, "réponse inattendue du daemon (tableau attendu)");
            return list;
        }

        public void Dispose() { client.Dispose(); }
    }
```

```csharp
using System;
using System.Collections.Generic;
using System.Text;

namespace IrisCli
{
    /// <summary>Rendu lisible (français) : durées, dates relatives, états, tableaux alignés.</summary>
    public static class TextFormatter
    {
        private static readonly long[] Sizes = { 86400, 3600, 60, 1 };
        private static readonly string[] Units = { "j", "h", "m", "s" };

        /// <summary>Les deux unités les plus significatives, sans unité nulle en queue : 12s, 5m30s, 1h, 1j1h.</summary>
        public static string Duration(long seconds)
        {
            if (seconds < 0) seconds = 0;
            for (int i = 0; i < Sizes.Length; i++)
            {
                long major = seconds / Sizes[i];
                if (major == 0 && i < Sizes.Length - 1) continue;
                string text = major + Units[i];
                if (i + 1 < Sizes.Length)
                {
                    long minor = (seconds % Sizes[i]) / Sizes[i + 1];
                    if (minor > 0) text += minor + Units[i + 1];
                }
                return text;
            }
            return "0s";
        }

        public static string Ago(DateTime then, DateTime now)
        {
            return "il y a " + Duration((long)Math.Floor((now.ToUniversalTime() - then.ToUniversalTime()).TotalSeconds));
        }

        public static string VaultState(string wire)
        {
            switch (wire)
            {
                case "absent": return "absent";
                case "locked": return "verrouillé";
                case "unlocked": return "déverrouillé";
                default: return wire ?? "?";
            }
        }

        public static string YesNo(bool value) { return value ? "oui" : "non"; }

        /// <summary>Colonnes séparées de deux espaces, dernière colonne non complétée.</summary>
        public static IList<string> Table(string[] headers, IList<string[]> rows)
        {
            int[] widths = new int[headers.Length];
            for (int i = 0; i < headers.Length; i++) widths[i] = headers[i].Length;
            foreach (string[] row in rows)
                for (int i = 0; i < row.Length; i++) widths[i] = Math.Max(widths[i], row[i].Length);
            List<string> lines = new List<string>();
            lines.Add(Row(headers, widths));
            foreach (string[] row in rows) lines.Add(Row(row, widths));
            return lines;
        }

        private static string Row(string[] cells, int[] widths)
        {
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < cells.Length; i++)
            {
                if (i < cells.Length - 1) sb.Append(cells[i].PadRight(widths[i] + 2));
                else sb.Append(cells[i]);
            }
            return sb.ToString();
        }
    }
```

```csharp
using System;
using System.IO;
using System.Runtime.InteropServices;
using System.Security;
using System.Text;

namespace IrisCli
{
    /// <summary>Valeurs de secrets (spec phase 5 §5) : lues en octets, jamais affichées ; les tampons intermédiaires sont effacés.</summary>
    public static class SecretValue
    {
        public static byte[] ReadAll(Stream input)
        {
            MemoryStream m = new MemoryStream();
            input.CopyTo(m);
            byte[] bytes = m.ToArray();
            Array.Clear(m.GetBuffer(), 0, (int)m.Length);
            return bytes;
        }

        /// <summary>Retire au plus un `\r\n` ou un `\n` final ; vide ou fait seulement d'espaces → code 1. `raw` est effacé.</summary>
        public static byte[] Normalize(byte[] raw)
        {
            int length = raw.Length;
            if (length >= 2 && raw[length - 2] == 13 && raw[length - 1] == 10) length -= 2;
            else if (length >= 1 && raw[length - 1] == 10) length -= 1;
            byte[] value = new byte[length];
            Buffer.BlockCopy(raw, 0, value, 0, length);
            Array.Clear(raw, 0, raw.Length);
            if (length == 0 || IsBlank(value))
            {
                Array.Clear(value, 0, value.Length);
                throw new CliException(CliExit.Logic, "la valeur ne peut pas être vide ni faite seulement d'espaces");
            }
            return value;
        }

        /// <summary>Blanche = UTF-8 valide fait seulement d'espaces ; des octets non UTF-8 ne sont jamais blancs (valeur binaire).</summary>
        private static bool IsBlank(byte[] value)
        {
            string text;
            try { text = new UTF8Encoding(false, true).GetString(value); }
            catch (DecoderFallbackException) { return false; }
            foreach (char c in text)
                if (!char.IsWhiteSpace(c)) return false;
            return true;
        }

        /// <summary>Saisie console → octets UTF-8 ; la copie en caractères est effacée.</summary>
        public static byte[] FromSecure(SecureString s)
        {
            char[] chars = Chars(s);
            try { return Encoding.UTF8.GetBytes(chars); }
            finally { Array.Clear(chars, 0, chars.Length); }
        }

        /// <summary>SecureString → string pour la trame JSON (limite du runtime acceptée en phase 3).</summary>
        public static string ToPlain(SecureString s)
        {
            char[] chars = Chars(s);
            try { return new string(chars); }
            finally { Array.Clear(chars, 0, chars.Length); }
        }

        public static bool SameText(SecureString a, SecureString b)
        {
            char[] x = Chars(a);
            char[] y = Chars(b);
            try
            {
                if (x.Length != y.Length) return false;
                int diff = 0;
                for (int i = 0; i < x.Length; i++) diff |= x[i] ^ y[i];
                return diff == 0;
            }
            finally
            {
                Array.Clear(x, 0, x.Length);
                Array.Clear(y, 0, y.Length);
            }
        }

        private static char[] Chars(SecureString s)
        {
            IntPtr bstr = Marshal.SecureStringToBSTR(s);
            try
            {
                char[] chars = new char[s.Length];
                Marshal.Copy(bstr, chars, 0, chars.Length);
                return chars;
            }
            finally { Marshal.ZeroFreeBSTR(bstr); }
        }
    }

    public static class CliInput
    {
        /// <summary>Une ligne UTF-8 sans `\r\n` ; null en fin de flux. Octet par octet : ne consomme rien au-delà de la ligne.</summary>
        public static string ReadLine(Stream input)
        {
            MemoryStream m = new MemoryStream();
            int b;
            while ((b = input.ReadByte()) >= 0 && b != '\n') m.WriteByte((byte)b);
            if (b < 0 && m.Length == 0) return null;
            return Encoding.UTF8.GetString(m.ToArray()).TrimEnd('\r');
        }
    }
}
```

```csharp
        private static int Add(ParsedArgs a, CliContext ctx)
        {
```

```csharp
using System;
using System.Globalization;
using System.Text.RegularExpressions;
using IrisKit;

namespace IrisCli
{
    /// <summary>`--since` / `--until` : `<entier><s|m|h|d>` soustrait de maintenant, ou date ISO-8601 ; rendu en UTC.</summary>
    public static class RelativeTime
    {
        private static readonly Regex Relative = new Regex("^([0-9]{1,9})([smhd])$", RegexOptions.CultureInvariant);

        public static DateTime Parse(string text, DateTime now)
        {
            Match m = Relative.Match(text ?? "");
            if (m.Success)
            {
                long n = long.Parse(m.Groups[1].Value, CultureInfo.InvariantCulture);
                switch (m.Groups[2].Value)
                {
                    case "s": return now.ToUniversalTime().AddSeconds(-n);
                    case "m": return now.ToUniversalTime().AddMinutes(-n);
                    case "h": return now.ToUniversalTime().AddHours(-n);
                    default: return now.ToUniversalTime().AddDays(-n);
                }
            }
            try { return Json.ParseDate(text ?? "").ToUniversalTime(); }
            catch (JsonException) { throw new UsageException("date invalide : " + text + " (attendu : 30s, 5m, 1h, 2d ou ISO-8601)"); }
            catch (FormatException) { throw new UsageException("date invalide : " + text + " (attendu : 30s, 5m, 1h, 2d ou ISO-8601)"); }
        }
    }
}
```

```csharp
using System;
using System.Collections;
using System.Collections.Generic;
using System.Collections.Specialized;
using System.Globalization;
using IrisKit;

namespace IrisCli
{
    /// <summary>`iris config get|set|reload` : lecture, mutation (le daemon écrit) et rechargement de config.json.</summary>
    internal static class ConfigCommands
    {
        public static void Register(IList<CommandSpec> list)
        {
            list.Add(new CommandSpec { Path = "config get", Usage = "config get [--json]", Summary = "affiche la configuration chargée par le daemon", Flags = new[] { "--json" }, UsesDaemon = true, Run = Get });
            list.Add(new CommandSpec { Path = "config set", Usage = "config set KEY VALUE [--json]", Summary = "modifie une clé (ex. security.on_exfil_attempt block_only)", Positionals = new[] { "KEY", "VALUE" }, Flags = new[] { "--json" }, UsesDaemon = true, Run = Set });
            list.Add(new CommandSpec { Path = "config reload", Usage = "config reload [--json]", Summary = "relit config.json (les clés broker.* demandent un redémarrage)", Flags = new[] { "--json" }, UsesDaemon = true, Run = Reload });
        }

        private static int Get(ParsedArgs a, CliContext ctx)
        {
            object raw;
            using (CliConnection c = CliConnection.Open(a, ctx)) raw = c.Call("config.get", null);
            if (a.Json)
            {
                CliOutput.Emit(ctx, raw);
                return CliExit.Ok;
            }
            CliConnection.AsObject(raw); // valide que la réponse est un objet
            Render(ctx, (IDictionary<string, object>)raw);
            return CliExit.Ok;
        }

        /// <summary>Rendu par sections : scalaires de tête, `[section]` par objet, `[[hosts]]` par hôte, autres tableaux en JSON compact.</summary>
        private static void Render(CliContext ctx, IDictionary<string, object> config)
        {
            foreach (KeyValuePair<string, object> kv in config)
                if (!(kv.Value is IDictionary<string, object>) && !(kv.Value is IList<object>)) ctx.Out.WriteLine(kv.Key + " = " + Scalar(kv.Value));
            foreach (KeyValuePair<string, object> kv in config)
            {
                IDictionary<string, object> section = kv.Value as IDictionary<string, object>;
                if (section == null) continue;
                ctx.Out.WriteLine();
                ctx.Out.WriteLine("[" + kv.Key + "]");
                foreach (KeyValuePair<string, object> field in section) ctx.Out.WriteLine(field.Key + " = " + Scalar(field.Value));
            }
            foreach (KeyValuePair<string, object> kv in config)
            {
                IList<object> items = kv.Value as IList<object>;
                if (items == null || items.Count == 0) continue;
                if (kv.Key != "hosts")
                {
                    ctx.Out.WriteLine();
                    ctx.Out.WriteLine(kv.Key + " = " + Json.EmitCompact(items));
                    continue;
                }
                foreach (object item in items)
                {
                    ctx.Out.WriteLine();
                    ctx.Out.WriteLine("[[hosts]]");
                    IDictionary<string, object> host = item as IDictionary<string, object>;
                    if (host == null) continue;
                    foreach (KeyValuePair<string, object> field in host) ctx.Out.WriteLine(field.Key + " = " + Scalar(field.Value));
                }
            }
        }

        private static string Scalar(object value)
        {
            if (value == null) return "null";
            string s = value as string;
            if (s != null) return "\"" + s + "\"";
            if (value is bool) return (bool)value ? "true" : "false";
            if (value is long || value is double) return Convert.ToString(value, CultureInfo.InvariantCulture);
            return Json.EmitCompact(value);
        }

        private static int Set(ParsedArgs a, CliContext ctx)
        {
            OrderedDictionary update = new OrderedDictionary();
            update["key"] = a.Positional(0);
            update["value"] = a.Positional(1);
            OrderedDictionary p = new OrderedDictionary();
            p["updates"] = new List<object> { update };
            object raw;
            using (CliConnection c = CliConnection.Open(a, ctx)) raw = c.Call("config.set", p);
            if (a.Json)
            {
                CliOutput.Emit(ctx, raw);
                return CliExit.Ok;
            }
            JsonObject r = CliConnection.AsObject(raw);
            IList<string> applied = r.StringArray("applied");
            IList<string> restart = r.StringArray("requires_restart");
            if (applied.Count > 0) ctx.Out.WriteLine("appliqué : " + string.Join(", ", applied));
            if (restart.Count > 0) ctx.Out.WriteLine("redémarrage requis : " + string.Join(", ", restart));
            return CliExit.Ok;
        }

        private static int Reload(ParsedArgs a, CliContext ctx)
        {
            object raw;
            using (CliConnection c = CliConnection.Open(a, ctx)) raw = c.Call("config.reload", null);
            IList<string> ignored = CliConnection.AsObject(raw).StringArray("ignored");
            if (ignored.Count > 0) ctx.Err.WriteLine("avertissement : modifications ignorées, redémarrage requis : " + string.Join(", ", ignored));
            if (a.Json) CliOutput.Emit(ctx, raw);
            else ctx.Out.WriteLine("configuration rechargée");
            return CliExit.Ok;
```

```csharp
using System;
using System.Collections.Generic;
using System.Collections.Specialized;
using System.Globalization;
using System.IO;
using System.Net.Sockets;
using IrisKit;

namespace IrisCli
{
    /// <summary>`iris logs` : `events.query` ponctuel, ou flux SSE (`--follow`) jusqu'à Ctrl+C ; ndjson en --json --follow.</summary>
    internal static class LogsCommand
    {
        public static void Register(IList<CommandSpec> list)
        {
            list.Add(new CommandSpec
            {
                Path = "logs",
                Usage = "logs [--since T] [--until T] [--limit N] [--kind k1,k2] [--host H] [--follow] [--json]",
                Summary = "events du daemon (T : 30s, 5m, 1h, 2d ou ISO-8601) ; --follow suit le flux jusqu'à Ctrl+C",
                Options = new[] { "--since", "--until", "--limit", "--kind", "--host" },
                Flags = new[] { "--follow", "--json" },
                UsesDaemon = true,
                Run = Run
            });
        }

        /// <summary>`<date ISO>  <kind>  <hôte>  <méthode> <chemin>  sub=<noms> dur=<ms>` (comme IRIS).</summary>
        public static string Line(Event e)
        {
            string subs = e.SubstitutedSecrets.Count == 0 ? "-" : string.Join(",", e.SubstitutedSecrets);
            string duration = e.DurationMs.HasValue ? e.DurationMs.Value.ToString(CultureInfo.InvariantCulture) + "ms" : "-";
            return Json.FormatDate(e.Timestamp) + "  " + EventWire.KindToWire(e.Kind) + "  " + e.Host + "  " + e.Method + " " + e.Path + "  sub=" + subs + " dur=" + duration;
        }

        private static int Run(ParsedArgs a, CliContext ctx)
        {
            List<string> kinds = Kinds(a.Option("--kind"));
            string host = a.Option("--host");
            if (a.Flag("--follow"))
            {
                if (a.Option("--since") != null || a.Option("--until") != null || a.Option("--limit") != null)
                    throw new UsageException("--follow est incompatible avec --since, --until et --limit");
                return Follow(a, ctx, kinds, host);
            }
            OrderedDictionary p = new OrderedDictionary();
            if (a.Option("--since") != null) p["since"] = Json.FormatDate(RelativeTime.Parse(a.Option("--since"), ctx.Now()));
            if (a.Option("--until") != null) p["until"] = Json.FormatDate(RelativeTime.Parse(a.Option("--until"), ctx.Now()));
            if (a.Option("--limit") != null)
            {
                int limit;
                if (!int.TryParse(a.Option("--limit"), NumberStyles.None, CultureInfo.InvariantCulture, out limit) || limit <= 0)
                    throw new UsageException("--limit attend un entier positif");
                p["limit"] = (long)limit;
            }
            if (kinds.Count > 0) p["kind"] = kinds;
            if (host != null) p["host"] = host;
            object raw;
            using (CliConnection c = CliConnection.Open(a, ctx)) raw = c.Call("events.query", p);
            if (a.Json)
            {
                CliOutput.Emit(ctx, raw);
                return CliExit.Ok;
            }
            IList<object> events = CliConnection.AsList(raw);
            if (events.Count == 0) ctx.Out.WriteLine("aucun event");
            foreach (object item in events) ctx.Out.WriteLine(Line(Event.FromJson(CliConnection.AsObject(item))));
            return CliExit.Ok;
        }

        private static List<string> Kinds(string raw)
        {
            List<string> kinds = new List<string>();
            if (raw == null) return kinds;
            foreach (string part in raw.Split(','))
            {
                string k = part.Trim();
                if (k.Length == 0) continue;
                try { EventWire.ParseKind(k); }
                catch (JsonException) { throw new UsageException("kind inconnu : " + k + " (substituted, passThrough, noMatch, exfilBlocked, error, systemAlert)"); }
                kinds.Add(k);
            }
            return kinds;
        }

        private static int Follow(ParsedArgs a, CliContext ctx, List<string> kinds, string host)
        {
            ctx.Interruptible = true;
            int port;
            using (CliConnection c = CliConnection.Open(a, ctx))
                port = EventsPort(CliConnection.AsObject(c.Call("config.get", null)).Object("broker").String("events_listen"));
            if (ctx.Cancel.IsCancellationRequested) return CliExit.Ok;
            EventsClient sse;
            try { sse = EventsClient.ConnectAsync(port, Query(kinds, host), TimeSpan.FromSeconds(60)).GetAwaiter().GetResult(); }
            catch (SocketException ex) { throw new CliException(CliExit.Io, "flux d'events injoignable : " + ex.Message); }
            catch (IOException ex) { throw new CliException(CliExit.Io, "flux d'events injoignable : " + ex.Message); }
            catch (EventsClientException ex) { throw new CliException(CliExit.Io, "flux d'events refusé : " + ex.Message); }
            using (sse)
            using (ctx.Cancel.Register(delegate { sse.Dispose(); }))
            {
                while (true)
                {
                    SseMessage m;
                    try { m = sse.ReadAsync().GetAwaiter().GetResult(); }
                    catch (Exception ex)
                    {
                        if (ctx.Cancel.IsCancellationRequested) return CliExit.Ok;
                        if (ex is IOException || ex is ObjectDisposedException || ex is SocketException || ex is TimeoutException)
                            throw new CliException(CliExit.Io, "flux d'events interrompu : " + ex.Message);
                        throw;
                    }
                    if (m == null)
                    {
                        if (ctx.Cancel.IsCancellationRequested) return CliExit.Ok;
                        throw new CliException(CliExit.Io, "flux d'events interrompu : le daemon a fermé la connexion");
                    }
                    if (m.IsComment) continue;
                    if (m.EventName == "dropped")
                    {
                        ctx.Err.WriteLine("avertissement : events perdus, lecteur trop lent (" + m.Data + ")");
                        continue;
                    }
                    Event e = m.ToEvent();
                    ctx.Out.WriteLine(a.Json ? Json.EmitCompact(e.ToJson()) : Line(e));
                    ctx.Out.Flush();
                }
            }
        }

        private static int EventsPort(string listen)
        {
            int colon = listen.LastIndexOf(':');
            int port;
            if (colon < 0 || !int.TryParse(listen.Substring(colon + 1), NumberStyles.None, CultureInfo.InvariantCulture, out port) || port <= 0 || port > 65535)
                throw new CliException(CliExit.Logic, "broker.events_listen invalide : " + listen);
            return port;
        }

        private static string Query(List<string> kinds, string host)
        {
            List<string> parts = new List<string>();
            if (kinds.Count > 0) parts.Add("kind=" + Uri.EscapeDataString(string.Join(",", kinds)));
            if (host != null) parts.Add("host=" + Uri.EscapeDataString(host));
            return string.Join("&", parts);
        }
    }
}
```

```csharp
using System.Collections.Generic;
using System.Collections.Specialized;
using System.Security;
using IrisKit;

namespace IrisCli
{
    /// <summary>
    /// `iris vault status|unlock|create` (spec phase 5 §4, §5) : le mot de passe maître ne vient que d'une console, jamais d'une
    /// option ni de stdin ; aucune nouvelle tentative dans la CLI (le daemon applique son délai anti-force brute).
    /// </summary>
    internal static class VaultCommands
    {
        public static void Register(IList<CommandSpec> list)
        {
            list.Add(new CommandSpec { Path = "vault status", Usage = "vault status [--json]", Summary = "état du coffre : absent, locked ou unlocked", Flags = new[] { "--json" }, UsesDaemon = true, Run = Status });
            list.Add(new CommandSpec { Path = "vault unlock", Usage = "vault unlock [--json]", Summary = "déverrouille le coffre (mot de passe saisi à la console, sans écho)", Flags = new[] { "--json" }, UsesDaemon = true, Run = Unlock });
            list.Add(new CommandSpec { Path = "vault create", Usage = "vault create [--json]", Summary = "crée le coffre (mot de passe saisi deux fois à la console)", Flags = new[] { "--json" }, UsesDaemon = true, Run = Create });
        }

        private static int Status(ParsedArgs a, CliContext ctx)
        {
            object raw;
            using (CliConnection c = CliConnection.Open(a, ctx)) raw = c.Call("vault.status", null);
            if (a.Json)
            {
                CliOutput.Emit(ctx, raw);
                return CliExit.Ok;
            }
            JsonObject s = CliConnection.AsObject(raw);
            string path = s.OptionalString("path");
            ctx.Out.WriteLine("coffre : " + TextFormatter.VaultState(s.String("state")) + (path == null ? " (secrets en mémoire)" : " (" + path + ")"));
            return CliExit.Ok;
        }

        private static void RequireConsole(CliContext ctx)
        {
            if (ctx.IsStdinRedirected)
                throw new CliException(CliExit.Usage, "vault unlock|create exige une console : le mot de passe maître ne se lit jamais sur l'entrée standard");
        }

        private static string State(CliConnection c) { return CliConnection.AsObject(c.Call("vault.status", null)).String("state"); }

        private static SecureString Read(CliContext ctx, string prompt)
        {
            SecureString s = ctx.Passwords.ReadPassword(prompt, ctx.Cancel);
            if (s == null) throw new CliException(CliExit.Logic, "saisie interrompue : rien n'a été modifié");
            if (s.Length == 0)
            {
                s.Dispose();
                throw new CliException(CliExit.Logic, "le mot de passe ne peut pas être vide");
            }
            return s;
        }

        private static object CallWithPassword(CliConnection c, string method, SecureString password)
        {
            OrderedDictionary p = new OrderedDictionary();
            p["password"] = SecretValue.ToPlain(password);
            try { return c.Call(method, p); }
            finally { p["password"] = null; }
        }

        private static OrderedDictionary Unlocked()
        {
            OrderedDictionary o = new OrderedDictionary();
            o["state"] = "unlocked";
            return o;
        }

        private static int Unlock(ParsedArgs a, CliContext ctx)
        {
            RequireConsole(ctx);
            object raw;
            string message;
            using (CliConnection c = CliConnection.Open(a, ctx))
            {
                string state = State(c);
                if (state == "absent") throw new CliException(CliExit.Logic, "aucun coffre : iris vault create");
                if (state == "unlocked")
                {
                    raw = Unlocked();
                    message = "déjà déverrouillé";
                }
                else
                {
                    using (SecureString password = Read(ctx, "Mot de passe maître : ")) raw = CallWithPassword(c, "vault.unlock", password);
                    message = "coffre déverrouillé";
                }
            }
            if (a.Json) CliOutput.Emit(ctx, raw);
            else ctx.Out.WriteLine(message);
            return CliExit.Ok;
        }

        private static int Create(ParsedArgs a, CliContext ctx)
        {
            RequireConsole(ctx);
            object raw;
            using (CliConnection c = CliConnection.Open(a, ctx))
            {
                string state = State(c);
                if (state != "absent") throw new CliException(CliExit.Logic, "un coffre existe déjà" + (state == "locked" ? " : iris vault unlock" : ""));
                using (SecureString first = Read(ctx, "Nouveau mot de passe maître : "))
                using (SecureString second = Read(ctx, "Confirmez le mot de passe maître : "))
                {
                    if (!SecretValue.SameText(first, second)) throw new CliException(CliExit.Logic, "les deux saisies diffèrent : coffre non créé");
                    raw = CallWithPassword(c, "vault.create", first);
                }
            }
            if (a.Json) CliOutput.Emit(ctx, raw);
            else ctx.Out.WriteLine("coffre créé et déverrouillé");
            return CliExit.Ok;
        }
    }
}
```

**Interfaces:**
- Consumes: tâches 2 et 3 (`OrderedJsonParser`, `OrderedJsonWriter`, `CommentPosition`, `McpPatcher`, `McpPatchResult`, `WrappedPathsRegistry`, `WrappedPathsCorruptException`), tâche 4 (`CliConnection`, `CliOutput`, `Paths.ExpandHome`, `RpcMessages`, `CliExit`), `LogLevel`, `LogLevelWire.TryParse`, `Json.FormatDate`.
- Produces:
  - `enum McpWrapOutcome { Patched, AlreadyCompliant, DryRun, CommentsFound, ChangedDuringPatch, SkippedOwnWrite }`.
  - `sealed class McpFileException : Exception` (`bool IsIo` : vrai → code 3, faux → code 1).
  - `sealed class McpWrapOptions` (champs `Path`, `BrokerListen`, `CaPemPath`, `DryRun`, `Registry`, `SkipIfHash`, `BeforeReplace`).
  - `sealed class McpWrapResult` (champs `Outcome`, `Patch`, `FirstComment`, `BackupPath`, `ContentHash`, `RegistryWarning`).
  - `McpWrapper.Wrap(McpWrapOptions)`, `McpWrapper.Unwrap(string path, WrappedPathsRegistry registry)` (rend un avertissement de registre ou null), `McpWrapper.BackupPathFor(string)`, `McpWrapper.Sha256(byte[])`, `McpWrapper.SameBytes(byte[], byte[])`.
  - `sealed class McpFileWatcher(string path, TimeSpan debounce) : IDisposable` avec `void WaitForChange(CancellationToken)` (OperationCanceledException à l'annulation).
  - Commandes `mcp wrap PATH [--dry-run] [--watch] [--log-level L] [--json]` et `mcp unwrap PATH [--json]`.

Deux précisions par rapport à la spec §6, à reporter dans le compte rendu de phase :
- `Unwrap` vit dans `McpWrapper` (E/S) et non dans `McpPatcher`, qui reste une fonction pure (la spec §6.1 le décrit ainsi).
- `--watch`, comme le Swift : contrôles préalables (backup → 1, fichier absent → 1), puis seul un daemon injoignable au **premier** cycle fait sortir (code 2) ; les autres échecs du premier cycle (JSON invalide, commentaires, fichier modifié) sont journalisés et la surveillance continue.

```csharp
using System;
using System.IO;
using System.Security.Cryptography;
using System.Text;

namespace IrisKit
{
    public enum McpWrapOutcome { Patched, AlreadyCompliant, DryRun, CommentsFound, ChangedDuringPatch, SkippedOwnWrite }

    /// <summary>Fichier MCP inutilisable : `IsIo` = échec d'écriture (code 3), sinon absent, illisible ou invalide (code 1).</summary>
    public sealed class McpFileException : Exception
    {
        public bool IsIo { get; private set; }

        public McpFileException(bool isIo, string message) : base(message) { IsIo = isIo; }
    }

    public sealed class McpWrapOptions
    {
        /// <summary>Chemin absolu du fichier MCP.</summary>
        public string Path;
        public string BrokerListen;
        public string CaPemPath;
        public bool DryRun;
        /// <summary>null : pas d'enregistrement.</summary>
        public WrappedPathsRegistry Registry;
        /// <summary>--watch : contenu déjà traité (écriture d'iris) → SkippedOwnWrite sans rien faire.</summary>
        public byte[] SkipIfHash;
        /// <summary>Tests : simule une écriture concurrente juste avant le remplacement.</summary>
        public Action BeforeReplace;
    }

    public sealed class McpWrapResult
    {
        public McpWrapOutcome Outcome;
        public McpPatchResult Patch;
        public CommentPosition FirstComment;
        public string BackupPath;
        /// <summary>SHA-256 du contenu laissé sur disque (écrit, ou lu s'il était déjà conforme).</summary>
        public byte[] ContentHash;
        public string RegistryWarning;
    }

    /// <summary>
    /// Un cycle de `iris mcp wrap` (spec phase 5 §6.2) : lecture, empreinte, parse JSONC, refus des commentaires, patch, backup des
    /// octets d'origine, revalidation stricte, contrôle d'écriture concurrente, remplacement atomique, registre. BOM et CRLF conservés.
    /// </summary>
    public static class McpWrapper
    {
        private static readonly UTF8Encoding StrictUtf8 = new UTF8Encoding(false, true);
        private static readonly byte[] Bom = { 0xEF, 0xBB, 0xBF };

        public static string BackupPathFor(string path) { return path + ".iris.bak"; }

        public static McpWrapResult Wrap(McpWrapOptions o)
        {
            McpWrapResult r = new McpWrapResult();
            r.BackupPath = BackupPathFor(o.Path);
            byte[] original = ReadFile(o.Path);
            byte[] readHash = Sha256(original);
            r.ContentHash = readHash;
            if (o.SkipIfHash != null && SameBytes(readHash, o.SkipIfHash))
            {
                r.Outcome = McpWrapOutcome.SkippedOwnWrite;
                return r;
            }
            bool bom, crlf;
            OrderedJsonDocument doc = ParseFile(o.Path, original, out bom, out crlf);
            if (doc.Comments.Count > 0 && !o.DryRun)
            {
                r.Outcome = McpWrapOutcome.CommentsFound;
                r.FirstComment = doc.Comments[0];
                return r;
            }
            r.Patch = McpPatcher.Patch(doc.Root, o.BrokerListen, o.CaPemPath);
            string after = OrderedJsonWriter.Serialize(r.Patch.Root);
            if (after == OrderedJsonWriter.Serialize(doc.Root))
            {
                r.Outcome = McpWrapOutcome.AlreadyCompliant;
                return r;
            }
            if (o.DryRun)
            {
                r.Outcome = McpWrapOutcome.DryRun;
                return r;
            }
            try { File.WriteAllBytes(r.BackupPath, original); }
            catch (Exception ex)
            {
                if (IsIo(ex)) throw new McpFileException(true, "écriture du backup " + r.BackupPath + " impossible : " + ex.Message);
                throw;
            }
            try { OrderedJsonParser.Parse(after, false); }
            catch (OrderedJsonException ex) { throw new McpFileException(false, "le résultat du patch ne se relit pas en JSON strict (" + ex.Message + ") : rien n'a été écrit"); }
            byte[] output = Encode(crlf ? after.Replace("\n", "\r\n") : after, bom);
            if (o.BeforeReplace != null) o.BeforeReplace();
            if (!Unchanged(o.Path, readHash))
            {
                r.Outcome = McpWrapOutcome.ChangedDuringPatch;
                r.ContentHash = null;
                return r;
            }
            ReplaceAtomically(o.Path, output);
            r.ContentHash = Sha256(output);
            r.Outcome = McpWrapOutcome.Patched;
            if (o.Registry != null) r.RegistryWarning = Register(o.Registry, o.Path, true);
            return r;
        }

        /// <summary>Restaure `path` depuis `path.iris.bak` (le backup est consommé) ; rend un avertissement de registre ou null.</summary>
        public static string Unwrap(string path, WrappedPathsRegistry registry)
        {
            string backup = BackupPathFor(path);
            if (!File.Exists(backup)) throw new McpFileException(false, "aucun backup : " + backup);
            byte[] bytes = ReadFile(backup);
            bool bom, crlf;
            ParseFile(backup, bytes, out bom, out crlf); // invalide → McpFileException, rien n'est touché
            try
            {
                if (File.Exists(path)) File.Replace(backup, path, null);
                else File.Move(backup, path);
            }
            catch (Exception ex)
            {
                if (IsIo(ex)) throw new McpFileException(true, "restauration de " + path + " impossible : " + ex.Message);
                throw;
            }
            return registry == null ? null : Register(registry, path, false);
        }

        public static byte[] Sha256(byte[] data)
        {
            using (SHA256 sha = SHA256.Create()) return sha.ComputeHash(data);
        }

        public static bool SameBytes(byte[] a, byte[] b)
        {
            if (a == null || b == null || a.Length != b.Length) return false;
            for (int i = 0; i < a.Length; i++)
                if (a[i] != b[i]) return false;
            return true;
        }

        private static bool IsIo(Exception ex) { return ex is IOException || ex is UnauthorizedAccessException; }

        private static byte[] ReadFile(string path)
        {
            try { return File.ReadAllBytes(path); }
            catch (FileNotFoundException) { throw new McpFileException(false, "fichier introuvable : " + path); }
            catch (DirectoryNotFoundException) { throw new McpFileException(false, "fichier introuvable : " + path); }
            catch (Exception ex)
            {
                if (IsIo(ex)) throw new McpFileException(false, "lecture de " + path + " impossible : " + ex.Message);
                throw;
            }
        }

        private static OrderedJsonDocument ParseFile(string path, byte[] bytes, out bool bom, out bool crlf)
        {
            bom = bytes.Length >= 3 && bytes[0] == Bom[0] && bytes[1] == Bom[1] && bytes[2] == Bom[2];
            int skip = bom ? 3 : 0;
            string text;
            try { text = StrictUtf8.GetString(bytes, skip, bytes.Length - skip); }
            catch (DecoderFallbackException) { throw new McpFileException(false, path + " n'est pas encodé en UTF-8"); }
            crlf = text.Contains("\r\n");
            try { return OrderedJsonParser.Parse(text, true); }
            catch (OrderedJsonException ex) { throw new McpFileException(false, path + " : JSON invalide, " + ex.Message); }
        }

        private static byte[] Encode(string text, bool bom)
        {
            byte[] body = new UTF8Encoding(false).GetBytes(text);
            if (!bom) return body;
            byte[] all = new byte[body.Length + 3];
            Buffer.BlockCopy(Bom, 0, all, 0, 3);
            Buffer.BlockCopy(body, 0, all, 3, body.Length);
            return all;
        }

        /// <summary>Le fichier a-t-il toujours le contenu lu au début du cycle ? Disparu ou illisible = changé.</summary>
        private static bool Unchanged(string path, byte[] readHash)
        {
            try { return SameBytes(Sha256(File.ReadAllBytes(path)), readHash); }
            catch (Exception ex)
            {
                if (IsIo(ex)) return false;
                throw;
            }
        }

        private static void ReplaceAtomically(string path, byte[] bytes)
        {
            string tmp = path + ".iris.tmp";
            try
            {
                File.WriteAllBytes(tmp, bytes);
                if (File.Exists(path)) File.Replace(tmp, path, null);
                else File.Move(tmp, path);
            }
            catch (Exception ex)
            {
                try { if (File.Exists(tmp)) File.Delete(tmp); }
                catch (IOException) { }
                if (IsIo(ex)) throw new McpFileException(true, "écriture de " + path + " impossible : " + ex.Message);
                throw;
            }
        }

        private static string Register(WrappedPathsRegistry registry, string path, bool add)
        {
            try
            {
                if (add) registry.Add(path);
                else registry.Remove(path);
                return null;
            }
            catch (WrappedPathsCorruptException ex) { return ex.Message; }
            catch (Exception ex)
            {
                if (IsIo(ex)) return "registre des fichiers enrobés non mis à jour : " + ex.Message;
                throw;
            }
        }
    }
```

```csharp
using System;
using System.IO;
using System.Threading;
using System.Threading.Tasks;

namespace IrisKit
{
    /// <summary>
    /// Observe un fichier (FileSystemWatcher sur son dossier, filtré sur son nom) et signale un changement après `debounce` sans
    /// nouvel événement (500 ms pour `mcp wrap --watch`, comme le FSEvents d'IRIS). Un changement survenu pendant le traitement du
    /// précédent n'est jamais perdu : il complète l'attente suivante.
    /// </summary>
    public sealed class McpFileWatcher : IDisposable
    {
        private readonly FileSystemWatcher watcher;
        private readonly Timer timer;
        private readonly TimeSpan debounce;
        private readonly object gate = new object();
        private TaskCompletionSource<bool> pending = new TaskCompletionSource<bool>();

        public McpFileWatcher(string path, TimeSpan debounce)
        {
            this.debounce = debounce;
            timer = new Timer(Fire, null, Timeout.Infinite, Timeout.Infinite);
            watcher = new FileSystemWatcher(Path.GetDirectoryName(path), Path.GetFileName(path));
            watcher.NotifyFilter = NotifyFilters.LastWrite | NotifyFilters.FileName | NotifyFilters.Size | NotifyFilters.CreationTime;
            watcher.Changed += OnEvent;
            watcher.Created += OnEvent;
            watcher.Renamed += OnEvent;
            watcher.EnableRaisingEvents = true;
        }

        private void OnEvent(object sender, FileSystemEventArgs e)
        {
            lock (gate) timer.Change(debounce, Timeout.InfiniteTimeSpan);
        }

        private void Fire(object state)
        {
            TaskCompletionSource<bool> done;
            lock (gate)
            {
                done = pending;
                pending = new TaskCompletionSource<bool>();
            }
            done.TrySetResult(true);
        }

        /// <summary>Rend la main au prochain changement (après debounce) ; OperationCanceledException si `cancel` est déclenché.</summary>
        public void WaitForChange(CancellationToken cancel)
        {
            Task next;
            lock (gate) next = pending.Task;
            try { next.Wait(cancel); }
            catch (AggregateException ex) { throw ex.InnerException; }
        }

        public void Dispose()
        {
            watcher.EnableRaisingEvents = false;
            watcher.Dispose();
            timer.Dispose();
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Collections.Specialized;
using System.IO;
using IrisKit;

namespace IrisCli
{
    /// <summary>`iris mcp wrap|unwrap` (SPECS §16.1, spec phase 5 §6) : seul le fichier MCP est écrit, jamais config.json ni le coffre.</summary>
    internal static class McpCommands
    {
        public static void Register(IList<CommandSpec> list)
        {
            list.Add(new CommandSpec
            {
                Path = "mcp wrap",
                Usage = "mcp wrap PATH [--dry-run] [--watch] [--log-level L] [--json]",
                Summary = "ajoute proxy et CA aux serveurs MCP du fichier (.mcp.json, .claude.json, claude_desktop_config.json)",
                Positionals = new[] { "PATH" },
                Options = new[] { "--log-level" },
                Flags = new[] { "--dry-run", "--watch", "--json" },
                UsesDaemon = true,
                Run = Wrap
            });
            list.Add(new CommandSpec { Path = "mcp unwrap", Usage = "mcp unwrap PATH [--json]", Summary = "restaure le fichier depuis PATH.iris.bak", Positionals = new[] { "PATH" }, Flags = new[] { "--json" }, Run = Unwrap });
        }

        private sealed class Target
        {
            public string Listen;
            public string CaPem;
        }

        private static string ResolvePath(string raw, CliContext ctx)
        {
            try { return Paths.ExpandHome(raw, ctx.UserProfile); }
            catch (ArgumentException ex) { throw new UsageException("chemin invalide : " + raw + " (" + ex.Message + ")"); }
            catch (NotSupportedException ex) { throw new UsageException("chemin invalide : " + raw + " (" + ex.Message + ")"); }
        }

        private static Target FetchTarget(string pipe, CliContext ctx)
        {
            Target t = new Target();
            using (CliConnection c = CliConnection.Open(pipe, ctx))
            {
                t.Listen = CliConnection.AsObject(c.Call("config.get", null)).Object("broker").String("listen");
                t.CaPem = CliConnection.AsObject(c.Call("ca.export_path", null)).String("path");
            }
            if (t.CaPem.Length == 0) throw new CliException(CliExit.Logic, "CA absent : déverrouillez ou créez le coffre (iris vault unlock), puis relancez");
            return t;
        }

        private static McpWrapOptions Options(string path, Target t, bool dryRun, CliContext ctx)
        {
            McpWrapOptions o = new McpWrapOptions();
            o.Path = path;
            o.BrokerListen = t.Listen;
            o.CaPemPath = t.CaPem;
            o.DryRun = dryRun;
            o.Registry = WrappedPathsRegistry.ForDataDir(ctx.DataDirectory);
            return o;
        }

        private static int Wrap(ParsedArgs a, CliContext ctx)
        {
            string path = ResolvePath(a.Positional(0), ctx);
            bool watch = a.Flag("--watch");
            if (watch && a.Flag("--dry-run")) throw new UsageException("--watch et --dry-run sont exclusifs");
            if (a.Option("--log-level") != null && !watch) throw new UsageException("--log-level ne s'utilise qu'avec --watch");
            if (path.EndsWith(".iris.bak", StringComparison.OrdinalIgnoreCase)) throw new CliException(CliExit.Logic, "refus d'enrober un backup : " + path);
            if (watch) return Watch(a, ctx, path);
            Target t = FetchTarget(CliConnection.ResolvePipe(a, ctx), ctx);
            McpWrapResult r;
            try { r = McpWrapper.Wrap(Options(path, t, a.Flag("--dry-run"), ctx)); }
            catch (McpFileException ex) { throw new CliException(ex.IsIo ? CliExit.Io : CliExit.Logic, ex.Message); }
            if (r.Outcome == McpWrapOutcome.CommentsFound)
                throw new CliException(CliExit.Logic, path + " : commentaires détectés en L" + r.FirstComment.Line + ":" + r.FirstComment.Column + " : retirez-les ou utilisez --dry-run");
            if (r.Outcome == McpWrapOutcome.ChangedDuringPatch)
                throw new CliException(CliExit.Logic, "fichier modifié pendant le patch : relancez (" + path + ")");
            if (r.RegistryWarning != null) ctx.Err.WriteLine("avertissement : " + r.RegistryWarning);
            if (a.Json)
            {
                CliOutput.Emit(ctx, Summary(r.Patch.Summary));
                return CliExit.Ok;
            }
            if (r.Outcome == McpWrapOutcome.AlreadyCompliant) ctx.Out.WriteLine("déjà conforme : " + path);
            else if (r.Outcome == McpWrapOutcome.DryRun)
            {
                foreach (string addition in r.Patch.Additions) ctx.Out.WriteLine("+ " + addition);
                foreach (string server in r.Patch.SkippedServers) ctx.Out.WriteLine("ignoré (http/sse sans env) : " + server);
                ctx.Out.WriteLine("(simulation : rien n'a été écrit)");
            }
            else ctx.Out.WriteLine("patché : " + path + " (backup : " + r.BackupPath + ")");
            return CliExit.Ok;
        }

        private static OrderedDictionary Summary(McpPatchSummary s)
        {
            OrderedDictionary o = new OrderedDictionary();
            o["patched"] = (long)s.Patched;
            o["already_compliant"] = (long)s.AlreadyCompliant;
            o["skipped"] = (long)s.Skipped;
            o["errors"] = 0L;
            return o;
        }

        private static int Unwrap(ParsedArgs a, CliContext ctx)
        {
            string path = ResolvePath(a.Positional(0), ctx);
            string warning;
            try { warning = McpWrapper.Unwrap(path, WrappedPathsRegistry.ForDataDir(ctx.DataDirectory)); }
            catch (McpFileException ex) { throw new CliException(ex.IsIo ? CliExit.Io : CliExit.Logic, ex.Message); }
            if (warning != null) ctx.Err.WriteLine("avertissement : " + warning);
            if (a.Json)
            {
                OrderedDictionary o = new OrderedDictionary();
                o["ok"] = true;
                o["restored"] = path;
                CliOutput.Emit(ctx, o);
            }
            else ctx.Out.WriteLine("restauré : " + path);
            return CliExit.Ok;
        }

        // ---- --watch ------------------------------------------------------------------------

        private static int Watch(ParsedArgs a, CliContext ctx, string path)
        {
            LogLevel minimum = LogLevel.Info;
            string level = a.Option("--log-level");
            if (level != null && !LogLevelWire.TryParse(level, out minimum))
                throw new UsageException("niveau de log inconnu : " + level + " (trace | debug | info | warn | error)");
            if (!File.Exists(path)) throw new CliException(CliExit.Logic, "fichier introuvable : " + path);
            WatchLoop loop = new WatchLoop(ctx, minimum, path, CliConnection.ResolvePipe(a, ctx));
            ctx.Interruptible = true;
            // L'observateur est créé avant le premier cycle : un changement pendant ce cycle n'est pas perdu.
            using (McpFileWatcher watcher = new McpFileWatcher(path, TimeSpan.FromMilliseconds(500)))
            {
                if (!loop.Cycle(true)) return CliExit.Unreachable;
                loop.Log(LogLevel.Info, "surveillance de " + path + " (Ctrl+C pour arrêter)");
                while (true)
                {
                    try { watcher.WaitForChange(ctx.Cancel); }
                    catch (OperationCanceledException)
                    {
                        loop.Log(LogLevel.Info, "arrêt de la surveillance");
                        return CliExit.Ok;
                    }
                    loop.Cycle(false);
                }
            }
        }

        /// <summary>Un cycle par changement : config et CA relus au daemon ; les échecs sont journalisés, la surveillance continue.</summary>
        private sealed class WatchLoop
        {
            private readonly CliContext ctx;
            private readonly LogLevel minimum;
            private readonly string path;
            private readonly string pipe;
            private byte[] lastHash;

            public WatchLoop(CliContext ctx, LogLevel minimum, string path, string pipe)
            {
                this.ctx = ctx;
                this.minimum = minimum;
                this.path = path;
                this.pipe = pipe;
            }

            public void Log(LogLevel level, string message)
            {
                if (level < minimum) return;
                ctx.Err.WriteLine(Json.FormatDate(ctx.Now()) + " " + LogLevelWire.ToWire(level) + " " + message);
                ctx.Err.Flush();
            }

            /// <summary>Faux seulement au premier cycle, daemon injoignable ou identité refusée (code 2).</summary>
            public bool Cycle(bool first)
            {
                Target t;
                try { t = FetchTarget(pipe, ctx); }
                catch (DaemonUnreachableException ex)
                {
                    if (first)
                    {
                        ctx.Err.WriteLine(ex.Message);
                        return false;
                    }
                    Log(LogLevel.Warn, "daemon injoignable : nouvel essai au prochain changement");
                    return true;
                }
                catch (AdminServerIdentityException ex)
                {
                    if (first)
                    {
                        ctx.Err.WriteLine("erreur : " + ex.Message);
                        return false;
                    }
                    Log(LogLevel.Warn, ex.Message);
                    return true;
                }
                catch (CliException ex)
                {
                    Log(LogLevel.Warn, ex.Message);
                    return true;
                }
                catch (RpcException ex)
                {
                    Log(LogLevel.Warn, "erreur du daemon : " + RpcMessages.For(ex));
                    return true;
                }
                McpWrapOptions o = Options(path, t, false, ctx);
                o.SkipIfHash = lastHash;
                McpWrapResult r;
                try { r = McpWrapper.Wrap(o); }
                catch (McpFileException ex)
                {
                    Log(LogLevel.Warn, ex.Message);
                    return true;
                }
                switch (r.Outcome)
                {
                    case McpWrapOutcome.SkippedOwnWrite:
                        Log(LogLevel.Debug, "contenu déjà traité (écriture d'iris) : ignoré");
                        break;
                    case McpWrapOutcome.AlreadyCompliant:
                        lastHash = r.ContentHash;
                        Log(LogLevel.Debug, "déjà conforme");
                        break;
                    case McpWrapOutcome.CommentsFound:
                        Log(LogLevel.Warn, "commentaires détectés en L" + r.FirstComment.Line + ":" + r.FirstComment.Column + " : écriture refusée");
                        break;
                    case McpWrapOutcome.ChangedDuringPatch:
                        Log(LogLevel.Warn, "fichier modifié pendant le patch : nouvel essai au prochain changement");
                        break;
                    default:
                        lastHash = r.ContentHash;
                        Log(LogLevel.Info, "fichier patché (backup : " + Path.GetFileName(r.BackupPath) + ", serveurs patchés : " + r.Patch.Summary.Patched + ")");
                        if (r.RegistryWarning != null) Log(LogLevel.Warn, r.RegistryWarning);
                        break;
                }
                return true;
            }
        }
    }
}
```

```csharp
        public static bool IsInAnyRoot(string thumbprint)
        {
            return InStore(StoreLocation.CurrentUser, thumbprint) || InStore(StoreLocation.LocalMachine, thumbprint);
        }

        private static bool InStore(StoreLocation location, string thumbprint)
        {
            X509Store store = new X509Store(StoreName.Root, location);
            try
            {
                store.Open(OpenFlags.ReadOnly | OpenFlags.OpenExistingOnly);
                return store.Certificates.Find(X509FindType.FindByThumbprint, thumbprint, false).Count > 0;
            }
            catch (System.Security.Cryptography.CryptographicException) { return false; }
            finally { store.Close(); }
        }
```

```csharp
    public enum DoctorSeverity { Ok, Warn, Fail }

        public DoctorSeverity Severity { get; private set; }

            Severity = severity;
            Detail = detail;
        }

        public static string SeverityWire(DoctorSeverity s)
        {

    /// Contrôles de `iris doctor` (SPECS §16, spec phase 5 §7), sans pipe : la CLI leur passe ce qu'elle a lu du daemon ; environnement,
    /// profil, magasins et processus sont injectés pour les tests. Noms identiques à IRIS quand le contrôle y existe.
    /// </summary>
    public static class DoctorChecks
    {
        private static DoctorResult Ok(string name, string detail) { return new DoctorResult(name, DoctorSeverity.Ok, detail); }
        private static DoctorResult Warn(string name, string detail) { return new DoctorResult(name, DoctorSeverity.Warn, detail); }
        private static DoctorResult Fail(string name, string detail) { return new DoctorResult(name, DoctorSeverity.Fail, detail); }

        public static DoctorResult AdminPipe(string pipeName, string error)
        {
            if (error == null) return Ok("admin-pipe", "\\\\.\\pipe\\" + pipeName);
            return Fail("admin-pipe", error);
        }

        public static DoctorResult DaemonAlive(long pid, long uptimeS, Func<int, bool> isAlive)
        {
            string detail = "pid=" + pid + " uptime_s=" + uptimeS;
            return isAlive((int)pid) ? Ok("daemon-alive", detail) : Fail("daemon-alive", detail + " : processus introuvable");
        }

        public static bool IsProcessAlive(int pid)
        {
            try
            {
                using (Process p = Process.GetProcessById(pid)) return !p.HasExited;
            }
            catch (ArgumentException) { return false; }
            catch (InvalidOperationException) { return false; }

        {
            if (state == "unlocked") return Ok("vault-unlocked", "coffre déverrouillé, MITM actif");
            if (state == "absent") return Fail("vault-unlocked", "aucun coffre : iris vault create");
            return Fail("vault-unlocked", "MITM inactif : iris vault unlock");
        }

        public static DoctorResult CaCertPresent(string pemPath)
        {
            if (string.IsNullOrEmpty(pemPath)) return Fail("ca-cert-present", "ca.pem absent : déverrouillez ou créez le coffre");
            return File.Exists(pemPath) ? Ok("ca-cert-present", pemPath) : Fail("ca-cert-present", "absent : " + pemPath);

        public static DoctorResult CaTrusted(string pemPath, Func<string, bool> isTrusted)

                {
                    if (isTrusted(cert.Thumbprint)) return Ok("ca-trusted", "dans les autorités racines de confiance");

                }
            }

        }

        public static DoctorResult ShellEnvVars(IDictionary<string, string> env, string brokerListen, string pemPath)
        {
            string expectedProxy = "http://" + brokerListen;
            List<string> missing = new List<string>();
            List<string> wrong = new List<string>();
            string proxy = Get(env, "HTTPS_PROXY");
            if (proxy == null) missing.Add("HTTPS_PROXY");
            else if (!string.Equals(proxy.TrimEnd('/'), expectedProxy, StringComparison.OrdinalIgnoreCase))
                wrong.Add("HTTPS_PROXY = " + proxy + " (attendu " + expectedProxy + ")");

                wrong.Add("NODE_EXTRA_CA_CERTS = " + ca + " (attendu " + pemPath + ")");

            if (missing.Count > 0) parts.Add("absentes : " + string.Join(",", missing));
            parts.AddRange(wrong);

        private static string Get(IDictionary<string, string> env, string key)
        {
            string value;
            if (env == null || !env.TryGetValue(key, out value) || string.IsNullOrEmpty(value)) return null;
            return value;
        }

        private static bool SamePath(string a, string b)
        {
            try { return string.Equals(Path.GetFullPath(a), Path.GetFullPath(b), StringComparison.OrdinalIgnoreCase); }
            catch (ArgumentException) { return false; }
            catch (NotSupportedException) { return false; }
        }

        /// <summary>`GET /__iris_ping` en TCP brut (ni proxy système ni HTTPS_PROXY) ; attendu `200` et corps `ok\n`.</summary>
        public static DoctorResult ProxyPing(string brokerListen, TimeSpan timeout)
        {
            const string name = "proxy-ping";
            int colon = brokerListen.LastIndexOf(':');
            int port;
            if (colon <= 0 || !int.TryParse(brokerListen.Substring(colon + 1), NumberStyles.None, CultureInfo.InvariantCulture, out port) || port <= 0 || port > 65535)
                return Fail(name, "broker.listen invalide : " + brokerListen);
            string host = brokerListen.Substring(0, colon);
            try
            {
                using (TcpClient tcp = new TcpClient())
                {
                    if (!tcp.ConnectAsync(host, port).Wait(timeout)) return Fail(name, "pas de réponse de " + brokerListen + " en " + (int)timeout.TotalMilliseconds + " ms");
                    tcp.ReceiveTimeout = (int)timeout.TotalMilliseconds;
                    tcp.SendTimeout = (int)timeout.TotalMilliseconds;
                    NetworkStream s = tcp.GetStream();
                    byte[] request = Encoding.ASCII.GetBytes("GET /__iris_ping HTTP/1.1\r\nHost: " + brokerListen + "\r\nConnection: close\r\n\r\n");
                    s.Write(request, 0, request.Length);
                    MemoryStream response = new MemoryStream();
                    byte[] buffer = new byte[1024];
                    int n;
                    while (response.Length < 8192 && (n = s.Read(buffer, 0, buffer.Length)) > 0) response.Write(buffer, 0, n);
                    string text = Encoding.ASCII.GetString(response.ToArray());
                    if (text.StartsWith("HTTP/1.1 200", StringComparison.Ordinal) && text.EndsWith("\r\n\r\nok\n", StringComparison.Ordinal)) return Ok(name, "200 ok");
                    int eol = text.IndexOf('\r');
                    return Fail(name, "réponse inattendue : " + (eol > 0 ? text.Substring(0, eol) : text));
                }
            }
            catch (AggregateException ex) { return Fail(name, brokerListen + " injoignable : " + ex.InnerException.Message); }
            catch (SocketException ex) { return Fail(name, brokerListen + " injoignable : " + ex.Message); }
            catch (IOException ex) { return Fail(name, brokerListen + " : " + ex.Message); }
        }

        /// <summary>`apiKeyHelper` dans %USERPROFILE%\.claude\settings.json est incompatible avec iris (SPECS Annexe A.11).</summary>
        public static DoctorResult ClaudeApiKeyHelper(string userProfile)
        {
            const string name = "claude-apikeyhelper-absent";
            string path = Path.Combine(userProfile, ".claude", "settings.json");
            if (!File.Exists(path)) return Ok(name, "pas de " + path);

                if (doc.Root.IsObject && doc.Root.Get("apiKeyHelper") != null)
                    return Fail(name, "apiKeyHelper défini dans " + path + " : incompatible avec iris (SPECS Annexe A.11)");
                return Ok(name, "apiKeyHelper absent");
            }
            catch (OrderedJsonException ex) { return Warn(name, path + " illisible : " + ex.Message); }
            catch (IOException ex) { return Warn(name, path + " illisible : " + ex.Message); }
            catch (UnauthorizedAccessException ex) { return Warn(name, path + " illisible : " + ex.Message); }
        }
    }
```

```csharp
using System;
using System.Collections.Generic;
using IrisKit;

namespace IrisCli
{
    /// <summary>`iris doctor` : huit contrôles dans l'ordre ; code 2 si le pipe est injoignable, 1 si un contrôle échoue, sinon 0.</summary>
    internal static class DoctorCommand
    {
        public static void Register(IList<CommandSpec> list)
        {
            list.Add(new CommandSpec
            {
                Path = "doctor",
                Usage = "doctor [--json]",
                Summary = "diagnostic : pipe, daemon, coffre, CA, variables d'environnement, proxy, Claude Code",
                Flags = new[] { "--json" },
                UsesDaemon = true,
                Run = Run
            });
        }

        private static int Run(ParsedArgs a, CliContext ctx)
        {
            List<DoctorResult> results = new List<DoctorResult>();
            string pipe = CliConnection.ResolvePipe(a, ctx);
            CliConnection c;
            try { c = CliConnection.Open(pipe, ctx); }
            catch (DaemonUnreachableException ex)
            {
                results.Add(DoctorChecks.AdminPipe(pipe, ex.Message.Replace(Environment.NewLine, " ; ")));
                Emit(a, ctx, results);
                return CliExit.Unreachable;
            }
            catch (AdminServerIdentityException ex)
            {
                results.Add(DoctorChecks.AdminPipe(pipe, ex.Message));
                Emit(a, ctx, results);
                return CliExit.Unreachable;
            }
            JsonObject status;
            JsonObject config;
            string pem;
            using (c)
            {
                status = CliConnection.AsObject(c.Call("daemon.status", null));
                pem = CliConnection.AsObject(c.Call("ca.export_path", null)).String("path");
                config = CliConnection.AsObject(c.Call("config.get", null));
            }
            string listen = config.Object("broker").String("listen");
            results.Add(DoctorChecks.AdminPipe(pipe, null));
            results.Add(DoctorChecks.DaemonAlive(status.Int("pid"), status.Int("uptime_s"), DoctorChecks.IsProcessAlive));
            results.Add(DoctorChecks.VaultUnlocked(status.String("vault")));
            DoctorResult present = DoctorChecks.CaCertPresent(pem);
            results.Add(present);
            if (present.Severity == DoctorSeverity.Ok) results.Add(DoctorChecks.CaTrusted(pem, ctx.IsCaTrusted));
            results.Add(DoctorChecks.ShellEnvVars(ctx.EnvironmentVariables, listen, pem));
            results.Add(DoctorChecks.ProxyPing(listen, TimeSpan.FromSeconds(2)));
            results.Add(DoctorChecks.ClaudeApiKeyHelper(ctx.UserProfile));
            Emit(a, ctx, results);
            foreach (DoctorResult r in results)
                if (r.Severity == DoctorSeverity.Fail) return CliExit.Logic;
            return CliExit.Ok;
        }

        private static void Emit(ParsedArgs a, CliContext ctx, List<DoctorResult> results)
        {
            if (a.Json)
            {
                List<object> items = new List<object>();
                foreach (DoctorResult r in results) items.Add(r.ToJson());
                CliOutput.Emit(ctx, items);
                return;
            }
            foreach (DoctorResult r in results)
                ctx.Out.WriteLine(("[" + DoctorResult.SeverityWire(r.Severity) + "]").PadRight(8) + r.Name + " — " + r.Detail);
        }
    }
}
```
